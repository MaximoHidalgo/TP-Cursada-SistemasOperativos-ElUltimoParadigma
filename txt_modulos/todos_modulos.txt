#include "ciclo_instruccion.h"

void ejecutar_ciclo_de_instruccion(t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    int desalojar = 0;

    while (!desalojar) {

        /* 1. FETCH */ 
        char* instruccion_texto = fetch(contexto, ctx);
        
        if (instruccion_texto == NULL) {
            log_error(ctx->logger, "Fetch falló (probablemente Memoria se desconectó). Desalojando PID %d.", contexto->pid);
            break;
        }

        /* 2. DECODE */
        t_instruccion* instruccion_decodificada = decode(instruccion_texto);
        
        /* 3. EXECUTE */
        desalojar = execute(instruccion_decodificada, contexto, ctx);
        
        // Incrementamos el PC solo si no hubo desalojo y la instrucción NO fue un JNZ
        if (!desalojar && !string_equals_ignore_case(instruccion_decodificada->identificador, "JNZ")) {
            contexto->registros.PC++;
        }

        /* 4. CHECK INTERRUPT */
        if (!desalojar) {
            int hubo_interrupcion = chequear_interrupciones(ctx);
            if (hubo_interrupcion) {
                log_info(ctx->logger, "## Interrupción por Fin de Quantum recibida para PID %d", contexto->pid);
                desalojar = 1; // Obligamos a romper el ciclo de instrucción
                
                enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
                notificar_desalojo_a_scheduler(contexto->pid, FIN_DE_QUANTUM, ctx);
            }
        }

        // Limpiamos memoria (y también liberar la instrucción y sus parámetros)
        free(instruccion_texto);
        if (instruccion_decodificada->identificador) free(instruccion_decodificada->identificador);
        if (instruccion_decodificada->parametro1) free(instruccion_decodificada->parametro1);
        if (instruccion_decodificada->parametro2) free(instruccion_decodificada->parametro2);
        if (instruccion_decodificada->parametro3) free(instruccion_decodificada->parametro3);
        free(instruccion_decodificada);
    }
}

void notificar_desalojo_a_scheduler(uint32_t pid, int motivo_desalojo, t_cpu_context* ctx) {
    // 1. Creamos un paquete. Usamos el 'motivo_desalojo' directamente como el op_code
    t_paquete* paquete = crear_paquete();
    paquete->codigo_operacion = motivo_desalojo; 
    
    // 2. Agregamos únicamente el PID al paquete
    agregar_a_paquete(paquete, &pid, sizeof(uint32_t));
    
    // 3. Enviamos el paquete por el socket conectado al Scheduler
    enviar_paquete(paquete, ctx->conexiones->scheduler);
    
    // 4. Liberamos el paquete para evitar memory leaks
    eliminar_paquete(paquete);

    log_info(ctx->logger, "PID %d devuelto al Scheduler. Motivo (op_code): %d", pid, motivo_desalojo);
}

void notificar_syscall_sleep_a_scheduler(uint32_t pid, uint32_t tiempo_sleep, t_cpu_context* ctx) {
    t_paquete* paquete = crear_paquete();
    paquete->codigo_operacion = SYSCALL_SLEEP; 
    
    agregar_a_paquete(paquete, &pid, sizeof(uint32_t));
    agregar_a_paquete(paquete, &tiempo_sleep, sizeof(uint32_t)); // ¡Dato extra!
    
    enviar_paquete(paquete, ctx->conexiones->scheduler);
    eliminar_paquete(paquete);
}

int obtener_tamanio_registro(char* nombre_registro) {
    if (string_equals_ignore_case(nombre_registro, "AX") || string_equals_ignore_case(nombre_registro, "BX") || 
        string_equals_ignore_case(nombre_registro, "CX") || string_equals_ignore_case(nombre_registro, "DX")) return 1;
    if (string_equals_ignore_case(nombre_registro, "EAX") || string_equals_ignore_case(nombre_registro, "EBX") || 
        string_equals_ignore_case(nombre_registro, "ECX") || string_equals_ignore_case(nombre_registro, "EDX") ||
        string_equals_ignore_case(nombre_registro, "SI") || string_equals_ignore_case(nombre_registro, "DI")) return 4;
    return 0;
}

int mmu_traducir(t_cpu_context* ctx, t_contexto_ejecucion* contexto, int dir_logica, int tam_bytes, int tam_max_segmento, uint32_t* dir_fisica) {
    int num_segmento = floor((double)dir_logica / tam_max_segmento);
    int desplazamiento = dir_logica % tam_max_segmento;

    t_segmento* seg = NULL;
    if (contexto->tabla_segmentos != NULL) {
        for (int i = 0; i < list_size(contexto->tabla_segmentos); i++) {
            t_segmento* s = list_get(contexto->tabla_segmentos, i);
            if (s->id == num_segmento) {
                seg = s;
                break;
            }
        }
    }

    if (!seg) return -1; // SEG FAULT

    if (desplazamiento + tam_bytes > seg->tamanio) {
        return -1; // SEG FAULT
    }

    *dir_fisica = seg->base + desplazamiento;
    return 0;
}

t_contexto_ejecucion* pedir_contexto_a_memoria(int pid_recibido, t_cpu_context* ctx) {
    t_paquete* paquete_solicitud = crear_paquete();
    paquete_solicitud->codigo_operacion = SOLICITUD_CONTEXTO; 
    
    //Agregamos el PID al paquete y lo enviamos
    agregar_a_paquete(paquete_solicitud, &pid_recibido, sizeof(int));
    enviar_paquete(paquete_solicitud, ctx->conexiones->memoria);
    eliminar_paquete(paquete_solicitud);

    log_info(ctx->logger, "Solicitando contexto del PID %d a Kernel Memory...", pid_recibido);

    //Quedarnos esperando la respuesta de la Memoria
    int operacion = recibir_operacion(ctx->conexiones->memoria);

    if (operacion == ENVIO_CONTEXTO) {
        int size;
        void* buffer = recibir_buffer(&size, ctx->conexiones->memoria);

        //Deserializar el buffer recibido para armar el struct localmente
        t_contexto_ejecucion* contexto = malloc(sizeof(t_contexto_ejecucion));
        deserializar_contexto(buffer, size, contexto);
        free(buffer);

        int cant = contexto->tabla_segmentos ? list_size(contexto->tabla_segmentos) : 0;
        log_info(ctx->logger, "Contexto de ejecución del PID %d recibido exitosamente con %d segmentos.", contexto->pid, cant);
        for(int i=0; i<cant; i++) {
            t_segmento* s = list_get(contexto->tabla_segmentos, i);
            log_info(ctx->logger, "  -> Seg ID: %d, Base: %d, Tamanio: %d", s->id, s->base, s->tamanio);
        }

        return contexto;
    } else {
        log_error(ctx->logger, "Error: Se esperaba el contexto de ejecución, pero llegó la operación: %d", operacion);
        return NULL;
    }
}

void enviar_contexto_a_memoria(t_contexto_ejecucion* contexto, int conexion_memory) {
    t_paquete* paquete = crear_paquete();
    paquete->codigo_operacion = ACTUALIZACION_CONTEXTO;
    serializar_contexto(paquete, contexto);
    enviar_paquete(paquete, conexion_memory);
    eliminar_paquete(paquete);
}

char* fetch(t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    //Creamos un paquete y le cargamos el OP Code FETCH_INSTRUCCION
    t_paquete* paquete_solicitud = crear_paquete();
    paquete_solicitud->codigo_operacion = FETCH_INSTRUCCION; 
    
    enviar_peticion_fetch_instruccion(contexto->pid, contexto->registros.PC, ctx->conexiones->memoria);

    int op_code = recibir_operacion(ctx->conexiones->memoria);
    
    if (op_code == ENVIO_INSTRUCCION) {
        t_list* lista = recibir_paquete(ctx->conexiones->memoria);
        char* instruccion_texto = string_duplicate(list_get(lista, 0));
        list_destroy_and_destroy_elements(lista, free);
        return instruccion_texto;
    } else {
        log_error(ctx->logger, "Error: Se esperaba una instrucción, pero llegó la operación: %d", op_code);
        return NULL;
    }
}

t_instruccion* decode(char* instruccion_texto) {
    t_instruccion* instruccion = malloc(sizeof(t_instruccion));
    instruccion->parametro1 = NULL;
    instruccion->parametro2 = NULL;
    instruccion->parametro3 = NULL;

    char** instruccion_separada = string_split(instruccion_texto, " ");

    // El primer string siempre es el comando a ejecutar
    instruccion->identificador = string_duplicate(instruccion_separada[0]);

    if (instruccion_separada[1] != NULL) {
    instruccion->parametro1 = string_duplicate(instruccion_separada[1]);
        if (instruccion_separada[2] != NULL) {
        instruccion->parametro2 = string_duplicate(instruccion_separada[2]);
            if (instruccion_separada[3] != NULL) {
                instruccion->parametro3 = string_duplicate(instruccion_separada[3]);
            }
        }
    }

    // Liberamos el array de strings que generó las Commons para evitar memory leaks
    string_array_destroy(instruccion_separada);

    return instruccion;
}

int execute(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    log_info(ctx->logger, "## PID: %d - Ejecutando: %s - %s %s %s", 
             contexto->pid, 
             instruccion->identificador, 
             instruccion->parametro1 ? instruccion->parametro1 : "", 
             instruccion->parametro2 ? instruccion->parametro2 : "", 
             instruccion->parametro3 ? instruccion->parametro3 : "");

    t_instruction *inst = instruction_get_by_name(instruccion->identificador);
    
    if (inst != NULL) {
        return inst->instr(instruccion, contexto, ctx);
    } else {
        log_error(ctx->logger, "Instrucción desconocida: %s", instruccion->identificador);
        return 1; // Desalojar
    }
}

int chequear_interrupciones(t_cpu_context* ctx) {
    int op_code_espiado;
    
    // Miramos si hay datos en el socket de forma no bloqueante
    ssize_t bytes_recibidos = recv(ctx->conexiones->scheduler, &op_code_espiado, sizeof(int), MSG_PEEK | MSG_DONTWAIT);

    if (bytes_recibidos > 0) {
        int operacion = recibir_operacion(ctx->conexiones->scheduler);
        
        if (operacion == INTERRUPCION_QUANTUM) { 
            // Consumimos el payload del paquete para no dejar basura en el buffer del socket
            int size;
            void* buffer = recibir_buffer(&size, ctx->conexiones->scheduler);
            free(buffer);

            log_info(ctx->logger, "## Interrupción recibida");
            return 1;
        } else {
            log_warning(ctx->logger, "Se recibió un código inesperado durante el ciclo: %d", operacion);
        }
    } else if (bytes_recibidos == -1 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
        // No hay mensajes pendientes y la CPU sigue ejecutando tranquila.
        return 0;
    } else if (bytes_recibidos == 0) {
        log_error(ctx->logger, "El Kernel Scheduler se ha desconectado inesperadamente.");
        return 1; // Frenamos la CPU porque se cayó el servidor
    }
    return 0;
}

void* leer_memoria_distribuida_cpu(t_cpu_context* ctx, uint32_t dir_fisica, uint32_t tamanio) {
    void* buffer = malloc(tamanio);
    uint32_t leido = 0;
    
    while (leido < tamanio) {
        uint32_t dir_actual = dir_fisica + leido;
        t_memory_stick_info* stick_target = NULL;
        
        for (int i = 0; i < list_size(ctx->memory_sticks); i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            if (dir_actual >= stick->base_address && dir_actual < stick->base_address + stick->tamanio) {
                stick_target = stick;
                break;
            }
        }
        
        if (stick_target == NULL) {
            log_error(ctx->logger, "Error: Direccion fisica fuera de los limites de la memoria distribuida (CPU)");
            break;
        }
        
        uint32_t offset_en_stick = dir_actual - stick_target->base_address;
        uint32_t bytes_disponibles = stick_target->tamanio - offset_en_stick;
        uint32_t bytes_a_leer = (tamanio - leido < bytes_disponibles) ? (tamanio - leido) : bytes_disponibles;
        
        enviar_peticion_lectura(offset_en_stick, bytes_a_leer, stick_target->socket_fd);
        
        int r_op = recibir_operacion(stick_target->socket_fd);
        if (r_op == LECTURA) {
            int size_buf;
            void* res = recibir_buffer(&size_buf, stick_target->socket_fd);
            memcpy(buffer + leido, res, bytes_a_leer);
            free(res);
        } else {
            log_error(ctx->logger, "Error al leer del memory stick en CPU");
            break;
        }
        
        leido += bytes_a_leer;
    }
    
    return buffer;
}

void escribir_memoria_distribuida_cpu(t_cpu_context* ctx, uint32_t dir_fisica, uint32_t tamanio, void* buffer) {
    uint32_t escrito = 0;
    
    while (escrito < tamanio) {
        uint32_t dir_actual = dir_fisica + escrito;
        t_memory_stick_info* stick_target = NULL;
        
        for (int i = 0; i < list_size(ctx->memory_sticks); i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            if (dir_actual >= stick->base_address && dir_actual < stick->base_address + stick->tamanio) {
                stick_target = stick;
                break;
            }
        }
        
        if (stick_target == NULL) {
            log_error(ctx->logger, "Error: Direccion fisica fuera de los limites de la memoria distribuida (CPU)");
            break;
        }
        
        uint32_t offset_en_stick = dir_actual - stick_target->base_address;
        uint32_t bytes_disponibles = stick_target->tamanio - offset_en_stick;
        uint32_t bytes_a_escribir = (tamanio - escrito < bytes_disponibles) ? (tamanio - escrito) : bytes_disponibles;
        
        enviar_peticion_escritura(offset_en_stick, bytes_a_escribir, buffer + escrito, stick_target->socket_fd);
        
        int r_op = recibir_operacion(stick_target->socket_fd);
        if (r_op == MENSAJE) {
            int size_buf;
            void* res = recibir_buffer(&size_buf, stick_target->socket_fd);
            free(res); // Esperamos MSG_OK
        } else {
            log_error(ctx->logger, "Error al escribir en memory stick en CPU");
            break;
        }
        
        escrito += bytes_a_escribir;
    }
}

t_instruction *instruction_get_by_name(char *name) {
    for (uint8_t i = 0; INSTRUCTION_SET[i].name != NULL; i++) {
        if (string_equals_ignore_case(INSTRUCTION_SET[i].name, name)) {
            return &INSTRUCTION_SET[i];
        }
    }
    return NULL;
}

int instr_set(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t valor = atoi(instruccion->parametro2);
    asignar_valor_a_registro(contexto, instruccion->parametro1, valor);
    return 0; // 0 significa no desalojar
}

int instr_sum(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t destino_val = leer_registro(contexto, instruccion->parametro1);
    uint32_t origen_val = leer_registro(contexto, instruccion->parametro2);
    asignar_valor_a_registro(contexto, instruccion->parametro1, destino_val + origen_val);
    return 0;
}

int instr_sub(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t destino_val = leer_registro(contexto, instruccion->parametro1);
    uint32_t origen_val = leer_registro(contexto, instruccion->parametro2);
    asignar_valor_a_registro(contexto, instruccion->parametro1, destino_val - origen_val);
    return 0;
}

int instr_jnz(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t registro_val = leer_registro(contexto, instruccion->parametro1);

    if (registro_val != 0) {
        contexto->registros.PC = atoi(instruccion->parametro2);
    } else {
        contexto->registros.PC++;
    }
    return 0;
}

int instr_mov_in(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    char* reg_datos = instruccion->parametro1;
    int tam = obtener_tamanio_registro(reg_datos);
    uint32_t dir_logica = contexto->registros.SI;
    uint32_t dir_fisica = 0;

    if (mmu_traducir(ctx, contexto, dir_logica, tam, ctx->tam_max_segmento, &dir_fisica) == -1) {
        log_error(ctx->logger, "Segmentation Fault accediendo a dir logica %d", dir_logica);
        enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = SYSCALL_SEG_FAULT;
        agregar_a_paquete(paq, &(contexto->pid), sizeof(uint32_t));
        enviar_paquete(paq, ctx->conexiones->scheduler);
        eliminar_paquete(paq);
        return 1; // Desalojar
    } else {
        void* buf = leer_memoria_distribuida_cpu(ctx, dir_fisica, tam);
        if (buf != NULL) {
            uint32_t valor = 0;
            memcpy(&valor, buf, tam);
            free(buf);
            asignar_valor_a_registro(contexto, reg_datos, valor);
            log_info(ctx->logger, "PID: %d - Acción: LEER - Dirección Física: %d - Valor: %d", contexto->pid, dir_fisica, valor);
        }
        return 0;
    }
}

int instr_mov_out(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    char* reg_datos = instruccion->parametro1;
    int tam = obtener_tamanio_registro(reg_datos);
    uint32_t valor = leer_registro(contexto, reg_datos);
    uint32_t dir_logica = contexto->registros.DI;
    uint32_t dir_fisica = 0;

    if (mmu_traducir(ctx, contexto, dir_logica, tam, ctx->tam_max_segmento, &dir_fisica) == -1) {
        log_error(ctx->logger, "Segmentation Fault escribiendo a dir logica %d", dir_logica);
        enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = SYSCALL_SEG_FAULT;
        agregar_a_paquete(paq, &(contexto->pid), sizeof(uint32_t));
        enviar_paquete(paq, ctx->conexiones->scheduler);
        eliminar_paquete(paq);
        return 1; // Desalojar
    } else {
        escribir_memoria_distribuida_cpu(ctx, dir_fisica, tam, &valor);
        log_info(ctx->logger, "PID: %d - Acción: ESCRIBIR - Dirección Física: %d - Valor: %d", contexto->pid, dir_fisica, valor);
        return 0;
    }
}

int instr_copy_mem(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t tam = atoi(instruccion->parametro1);
    uint32_t dir_logica_si = contexto->registros.SI;
    uint32_t dir_logica_di = contexto->registros.DI;
    uint32_t dir_fisica_si = 0;
    uint32_t dir_fisica_di = 0;

    if (mmu_traducir(ctx, contexto, dir_logica_si, tam, ctx->tam_max_segmento, &dir_fisica_si) == -1 || 
        mmu_traducir(ctx, contexto, dir_logica_di, tam, ctx->tam_max_segmento, &dir_fisica_di) == -1) {
        log_error(ctx->logger, "Segmentation Fault en COPY_MEM");
        enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = SYSCALL_SEG_FAULT;
        agregar_a_paquete(paq, &(contexto->pid), sizeof(uint32_t));
        enviar_paquete(paq, ctx->conexiones->scheduler);
        eliminar_paquete(paq);
        return 1;
    } else {
        void* buf = leer_memoria_distribuida_cpu(ctx, dir_fisica_si, tam);
        if (buf != NULL) {
            escribir_memoria_distribuida_cpu(ctx, dir_fisica_di, tam, buf);
            log_info(ctx->logger, "PID: %d - Acción: COPY_MEM - Origen Fisico: %d - Destino Fisico: %d", contexto->pid, dir_fisica_si, dir_fisica_di);
            free(buf);
        }
        return 0;
    }
}

    }
    
    usleep(1000 * 100); // 100ms provisorio, podés ajustarlo al RETARDO_INSTRUCCION si luego lo agregás al config
    
    for (int i = 1; i < repeticiones; i++) {
        // Simulamos la ejecución reiterada
        log_info(ctx->logger, "## PID: %d - Ejecutando: NOOP -   ", contexto->pid);
        usleep(1000 * 100); // 100ms de retardo por cada repetición simulada
    }
    
    return 0;
}

t_cpu_context* iniciar_contexto(int argc, char* argv[]) {
    validar_argumentos(argc);

    t_cpu_context* ctx = malloc(sizeof(t_cpu_context));
    if (!ctx) return NULL;

    ctx->id_cpu = strdup(argv[2]);
    char* path_config = argv[1];

    printf("Se arranco el CPU correctamente con el id: %s y el path del config: %s\n", ctx->id_cpu, path_config);

    char* nombre_log = string_from_format("%s.log", ctx->id_cpu);
    ctx->logger = log_create(nombre_log, "CPU", true, LOG_LEVEL_INFO);
    free(nombre_log);

    ctx->config = config_create(path_config);
    if (ctx->config == NULL) {
        log_error(ctx->logger, "No se pudo leer el archivo de configuracion");
        log_destroy(ctx->logger);
        free(ctx->id_cpu);
        free(ctx);
        return NULL;
    }

    char* ip_scheduler = config_get_string_value(ctx->config, "IP_SCHEDULER");
    char* puerto_scheduler = config_get_string_value(ctx->config, "PUERTO_SCHEDULER");
    char* ip_memory = config_get_string_value(ctx->config, "IP_MEMORY");
    char* puerto_memory = config_get_string_value(ctx->config, "PUERTO_MEMORY");
    
    ctx->conexiones = malloc(sizeof(t_conexiones_cpu));
    ctx->conexiones->sticks = -1;

    log_info(ctx->logger, "Intentando conectar Kernel Memory...");
    ctx->conexiones->memoria = crear_conexion(ip_memory, puerto_memory);
    ctx->memory_sticks = list_create();
    if (ctx->conexiones->memoria != -1) {
        log_info(ctx->logger, "## Conectado a Kernel Memory");
        enviar_mensaje(ctx->id_cpu, ctx->conexiones->memoria);

        int op = SOLICITUD_TAM_MAX_SEGMENTO;
        send(ctx->conexiones->memoria, &op, sizeof(int), 0);
        
        // El Kernel Memory va a responder con ENVIO_TABLA_STICKS y TAM_MAX_SEGMENTO
        int recibidos = 0;
        while (recibidos < 2) {
            int resp = recibir_operacion(ctx->conexiones->memoria);
            if (resp == TAM_MAX_SEGMENTO) {
                t_list* paq = recibir_paquete(ctx->conexiones->memoria);
                int* val = list_get(paq, 0);
                ctx->tam_max_segmento = *val;
                list_destroy_and_destroy_elements(paq, free);
                log_info(ctx->logger, "Tamaño maximo de segmento recibido: %d", ctx->tam_max_segmento);
                recibidos++;
            } else if (resp == ENVIO_TABLA_STICKS) {
                int size;
                void* buf = recibir_buffer(&size, ctx->conexiones->memoria);
                int desp = 0;
                int cant;
                desp += sizeof(int); // skip size of cant
                memcpy(&cant, buf + desp, sizeof(int)); desp += sizeof(int);
                
                for (int i = 0; i < cant; i++) {
                    t_memory_stick_info* stick = malloc(sizeof(t_memory_stick_info));
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->id), buf + desp, sizeof(int)); desp += sizeof(int);
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->tamanio), buf + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->base_address), buf + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->puerto), buf + desp, sizeof(int)); desp += sizeof(int);
                    
                    int ip_len;
                    desp += sizeof(int); // skip size
                    memcpy(&ip_len, buf + desp, sizeof(int)); desp += sizeof(int);
                    
                    desp += sizeof(int); // skip size
                    stick->ip = malloc(ip_len);
                    memcpy(stick->ip, buf + desp, ip_len); desp += ip_len;
                    
                    stick->socket_fd = crear_conexion(stick->ip, string_itoa(stick->puerto)); // TODO free itoa
                    if (stick->socket_fd != -1) {
                        enviar_mensaje(ctx->id_cpu, stick->socket_fd);
                        list_add(ctx->memory_sticks, stick);
                        log_info(ctx->logger, "Conectado a Memory Stick ID %d en %s:%d", stick->id, stick->ip, stick->puerto);
                    } else {
                        log_error(ctx->logger, "Error al conectar con Memory Stick ID %d", stick->id);
                    }
                }
                free(buf);
                recibidos++;
            } else {
                log_error(ctx->logger, "Operacion no esperada en handshake de memory: %d", resp);
                break;
            }
        }
    }

    log_info(ctx->logger, "Intentando conectar Kernel Scheduler...");
    ctx->conexiones->scheduler = crear_conexion(ip_scheduler, puerto_scheduler);
    if (ctx->conexiones->scheduler != -1) {
        log_info(ctx->logger, "## Conectado a Kernel Scheduler");
        enviar_mensaje(ctx->id_cpu, ctx->conexiones->scheduler);
    }

    return ctx;
}

void limpiar_contexto(t_cpu_context* ctx) {
    if (!ctx) return;
    
    // Si la limpieza vieja existe, la reusamos.
    if (ctx->conexiones) {
        limpieza_programa(ctx->conexiones, ctx->logger, ctx->config);
    } else {
        if (ctx->logger) log_destroy(ctx->logger);
        if (ctx->config) config_destroy(ctx->config);
    }
    
    if (ctx->id_cpu) free(ctx->id_cpu);
    free(ctx);
}

int main(int argc, char* argv[]) {
    t_cpu_context* ctx = iniciar_contexto(argc, argv);
    if (!ctx) return EXIT_FAILURE;

    while (1) {
        log_info(ctx->logger, "Esperando un nuevo PID del Kernel Scheduler...");
        int process_ID = recibir_operacion(ctx->conexiones->scheduler);
        
        switch(process_ID) {
            case PID_A_EJECUTAR:
            {
                int size;
                void* buffer = recibir_buffer(&size, ctx->conexiones->scheduler);
                
                int desplazamiento = 0;
                int size_payload;
                memcpy(&size_payload, buffer + desplazamiento, sizeof(int));
                desplazamiento += sizeof(int);
                
                int pid_recibido;
                memcpy(&pid_recibido, buffer + desplazamiento, sizeof(int));
                free(buffer);

                manejar_ejecucion_proceso(pid_recibido, ctx);
                break;
            }
            case -1:
                log_error(ctx->logger, "El Kernel Scheduler se desconectó inesperadamente.");
                printf("Termino la CPU y se cerro correctamente\n");
                limpiar_contexto(ctx);
                return EXIT_FAILURE;
            default:
                log_warning(ctx->logger, "Operación desconocida recibida del Scheduler: %d", process_ID);
                limpiar_contexto(ctx);
                return EXIT_FAILURE;
        }
    }

    limpiar_contexto(ctx);
    return 0;
}

t_io_context* iniciar_contexto(int argc, char* argv[]) {
    if (argc != 3){
        printf("Error en parametros. Uso: ./io <path_config> <nombre_io>\n");
        return NULL;
    }

    t_io_context* ctx = malloc(sizeof(t_io_context));
    if (ctx == NULL) return NULL;

    ctx->io_name = strdup(argv[2]);
    ctx->config = config_create(argv[1]);
    
    if (ctx->config == NULL){
        printf("No se pudo leer el archivo de config del modulo IO (%s)\n", argv[1]);
        free(ctx->io_name);
        free(ctx);
        return NULL;
    }

    char* ipKernel = config_get_string_value(ctx->config, "IP_SCHEDULER");
    char* puertoKernel = config_get_string_value(ctx->config, "PUERTO_SCHEDULER");
    char* log_level_str = config_get_string_value(ctx->config, "LOG_LEVEL");
    
    t_log_level nivelLog = log_level_from_string(log_level_str);
    ctx->logger = log_create("io.log", ctx->io_name, true, nivelLog);

    if (ctx->logger == NULL){
        printf("No se pudo crear el logger en modulo IO\n");
        config_destroy(ctx->config);
        free(ctx->io_name);
        free(ctx);
        return NULL;
    }

    ctx->socket_scheduler = crear_conexion(ipKernel, puertoKernel);

    if(ctx->socket_scheduler == -1){
        log_error(ctx->logger, "No se pudo establecer conexion con el Kernel Scheduler\n");
        log_destroy(ctx->logger);
        config_destroy(ctx->config);
        free(ctx->io_name);
        free(ctx);
        return NULL;
    }

    log_info(ctx->logger, "## Conectado a Kernel Scheduler");
    enviar_mensaje(ctx->io_name, ctx->socket_scheduler);
    log_info(ctx->logger, "Interfaz %s lista.", ctx->io_name);
    
    return ctx;
}

void limpiar_contexto(t_io_context* ctx) {
    if (ctx == NULL) return;
    
    if (ctx->socket_scheduler != -1) {
        liberar_conexion(ctx->socket_scheduler);
    }
    if (ctx->logger) log_destroy(ctx->logger);
    if (ctx->config) config_destroy(ctx->config);
    if (ctx->io_name) free(ctx->io_name);
    
    free(ctx);
}

int main(int argc, char* argv[]) {
    t_io_context* ctx = iniciar_contexto(argc, argv);
    if (ctx == NULL) {
        return EXIT_FAILURE;
    }

    while (1) {
        int cod_op = recibir_operacion(ctx->socket_scheduler);
        
        switch(cod_op) {
            case PETICION_IO:
                procesar_peticion_io(ctx);
                break;
            case -1:
                log_error(ctx->logger, "Se desconectó el Kernel Scheduler.");
                limpiar_contexto(ctx);
                return EXIT_FAILURE;
            default:
                log_warning(ctx->logger, "Operación desconocida: %d", cod_op);
                break;
        }
    }

    limpiar_contexto(ctx);
    return 0;
}

void ejecutar_stdin(t_io_context* ctx, int pid) {
    int cantidad_a_leer = recibir_int(ctx->socket_scheduler);
    
    log_info(ctx->logger, "## PID: %d - Ingrese %d caracteres:", pid, cantidad_a_leer);
    
    char* leido = readline("> ");
    char* texto_final = malloc(cantidad_a_leer + 1);

    int len_leido = strlen(leido);
    for (int i = 0; i < cantidad_a_leer; i++) {
        if (i < len_leido) {
            texto_final[i] = leido[i];
        } else {
            texto_final[i] = '\0';
        }
    }
    texto_final[cantidad_a_leer] = '\0';

    log_info(ctx->logger, "## Texto leido: %s", texto_final);

    // Enviar el texto leido al Kernel Scheduler
    enviar_mensaje(texto_final, ctx->socket_scheduler);

    free(leido);
    free(texto_final);
}

void ejecutar_stdout(t_io_context* ctx, int pid) {
    char* texto = recibir_string(ctx->socket_scheduler);
    
    log_info(ctx->logger, "## PID: %d - %s", pid, texto);
    
    free(texto);
}

void ejecutar_sleep(t_io_context* ctx, int pid) {
    int milisegundos = recibir_int(ctx->socket_scheduler); 

    log_info(ctx->logger, "## PID: %d - Haciendo sleep por %d milisegundos.", pid, milisegundos);
    
    usleep(milisegundos * 1000);
}

void procesar_peticion_io(t_io_context* ctx) {
    int pid = recibir_int(ctx->socket_scheduler); 
    
    log_info(ctx->logger, "## PID: %d - Inicio de IO", pid);

    if (string_equals_ignore_case(ctx->io_name, "SLEEP")) {
        ejecutar_sleep(ctx, pid);
        enviar_mensaje(MSG_OK, ctx->socket_scheduler);
    } 
    else if (string_equals_ignore_case(ctx->io_name, "STDOUT")) {
        ejecutar_stdout(ctx, pid);
        enviar_mensaje(MSG_OK, ctx->socket_scheduler);
    } 
    else if (string_equals_ignore_case(ctx->io_name, "STDIN")) {
        ejecutar_stdin(ctx, pid);
        // STDIN ya envía el texto leido en su funcion
    }

    log_info(ctx->logger, "## PID: %d - Fin de IO", pid);
}

t_programa* buscar_programa(t_memory_context* ctx, int pid) {
    pthread_mutex_lock(&ctx->mutex_programas);
    for (int i = 0; i < list_size(ctx->programas); i++) {
        t_programa* prog = list_get(ctx->programas, i);
        if (prog->pid == pid) {
            pthread_mutex_unlock(&ctx->mutex_programas);
            return prog;
        }
    }
    pthread_mutex_unlock(&ctx->mutex_programas);
    return NULL;
}

char** leer_lineas(FILE* f, int* cantidad) {
    char** lineas = NULL;
    char buffer[256];
    int count = 0;

    while (fgets(buffer, sizeof(buffer), f)) {
        lineas = realloc(lineas, sizeof(char*) * (count + 1));
        char* linea = strdup(buffer);
        string_trim(&linea);
        lineas[count] = linea;
        count++;
    }
    *cantidad = count;
    return lineas;
}

void crear_proceso(t_memory_context* ctx, int pid, char* path) {
    FILE* f = fopen(path, "r");

    if (!f) {
        // Fallback for tests
        char fallback_path[512];
        snprintf(fallback_path, sizeof(fallback_path), "../utils/scripts/%s", path);
        f = fopen(fallback_path, "r");
        
        if (!f) {
            log_error(ctx->logger, "No se pudo abrir el archivo %s ni %s", path, fallback_path);
            return;
        }
    }

    t_programa* prog = malloc(sizeof(t_programa));
    prog->pid = pid;
    prog->instrucciones = leer_lineas(f, &prog->cantidad_instrucciones);
    fclose(f);

    prog->contexto = malloc(sizeof(t_contexto_ejecucion));
    prog->contexto->pid = pid;
    memset(&(prog->contexto->registros), 0, sizeof(t_registros));
    prog->contexto->tabla_segmentos = list_create();

    pthread_mutex_lock(&ctx->mutex_programas);
    list_add(ctx->programas, prog);
    pthread_mutex_unlock(&ctx->mutex_programas);

    log_info(ctx->logger, "## PID: %d - Proceso Creado con %d instrucciones", pid, prog->cantidad_instrucciones);
}

char* obtener_instruccion(t_memory_context* ctx, int pid, int pc) {
    t_programa* prog = buscar_programa(ctx, pid);

    if (!prog || pc >= prog->cantidad_instrucciones)
        return strdup("EXIT");

    usleep(ctx->instruction_delay * 1000);

    char* instruccion = strdup(prog->instrucciones[pc]);

    log_info(ctx->logger, "## PID: %d - Obtener instrucción: %d - Instrucción: %s",
             pid, pc, instruccion);

    return instruccion;
}

t_hueco_libre* buscar_hueco_libre(t_memory_context* ctx, uint32_t tamanio) {
    t_hueco_libre* hueco_elegido = NULL;

    for (int i = 0; i < list_size(ctx->huecos_libres); i++) {
        t_hueco_libre* hueco = list_get(ctx->huecos_libres, i);
        uint32_t tam_hueco = hueco->limite - hueco->base;

        if (tam_hueco >= tamanio) {
            if (string_equals_ignore_case(ctx->allocation_strategy, "FIRST")) {
                return hueco;
            } else if (string_equals_ignore_case(ctx->allocation_strategy, "BEST")) {
                if (hueco_elegido == NULL || tam_hueco < (hueco_elegido->limite - hueco_elegido->base)) {
                    hueco_elegido = hueco;
                }
            } else if (string_equals_ignore_case(ctx->allocation_strategy, "WORST")) {
                if (hueco_elegido == NULL || tam_hueco > (hueco_elegido->limite - hueco_elegido->base)) {
                    hueco_elegido = hueco;
                }
            }
        }
    }
    return hueco_elegido;
}

t_segmento* crear_segmento(t_memory_context* ctx, uint32_t id, uint32_t tamanio, int* needs_compaction) {
    if (tamanio > ctx->segment_max_size) {
        log_error(ctx->logger, "Tamanio solicitado %d mayor a SEGMENT_MAX_SIZE %d", tamanio, ctx->segment_max_size);
        return NULL;
    }

    t_hueco_libre* hueco = buscar_hueco_libre(ctx, tamanio);
    
    if (hueco == NULL) {
        uint32_t memoria_libre_total = 0;
        for (int i = 0; i < list_size(ctx->huecos_libres); i++) {
            t_hueco_libre* h = list_get(ctx->huecos_libres, i);
            memoria_libre_total += (h->limite - h->base);
        }

        if (memoria_libre_total >= tamanio) {
            log_info(ctx->logger, "Espacio suficiente pero fragmentado. Compactacion requerida...");
            *needs_compaction = 1;
            return NULL;
        }
        
        log_error(ctx->logger, "Out of Memory para segmento %d de %d bytes", id, tamanio);
        return NULL; // OOM
    }

    t_segmento* segmento = malloc(sizeof(t_segmento));
    segmento->id = id;
    segmento->base = hueco->base;
    segmento->tamanio = tamanio;

    hueco->base += tamanio;
    if (hueco->base == hueco->limite) {
        for (int i = 0; i < list_size(ctx->huecos_libres); i++) {
            if (list_get(ctx->huecos_libres, i) == hueco) {
                list_remove_and_destroy_element(ctx->huecos_libres, i, free);
                break;
            }
        }
    }

    list_add(ctx->segmentos_globales, segmento);
    log_info(ctx->logger, "PID: <PID_FALTANTE> - Crear Segmento: %d - Base: %d - TAMAÑO: %d", id, segmento->base, tamanio);
    return segmento;
}

void eliminar_segmento(t_memory_context* ctx, t_segmento* segmento) {
    for (int i = 0; i < list_size(ctx->segmentos_globales); i++) {
        if (list_get(ctx->segmentos_globales, i) == segmento) {
            list_remove(ctx->segmentos_globales, i);
            break;
        }
    }

    t_hueco_libre* nuevo_hueco = malloc(sizeof(t_hueco_libre));
    nuevo_hueco->base = segmento->base;
    nuevo_hueco->limite = segmento->base + segmento->tamanio;
    list_add(ctx->huecos_libres, nuevo_hueco);

    log_info(ctx->logger, "PID: <PID_FALTANTE> - Eliminar Segmento: %d - Base: %d - TAMAÑO: %d", segmento->id, segmento->base, segmento->tamanio);
    free(segmento);
}

bool ordenar_segmentos_por_base(void* a, void* b) {
    t_segmento* seg1 = (t_segmento*)a;
    t_segmento* seg2 = (t_segmento*)b;
    return seg1->base < seg2->base;
}

void compactar_memoria(t_memory_context* ctx) {
    log_info(ctx->logger, "## Inicio de compactación");
    usleep(config_get_int_value(ctx->config, "COMPACTION_DELAY") * 1000);
    
    list_sort(ctx->segmentos_globales, ordenar_segmentos_por_base);
    
    uint32_t offset_actual = 0;
    for (int i = 0; i < list_size(ctx->segmentos_globales); i++) {
        t_segmento* seg = list_get(ctx->segmentos_globales, i);
        if (seg->base != offset_actual) {
            void* temp_buffer = leer_memoria_distribuida(ctx, seg->base, seg->tamanio);
            if (temp_buffer != NULL) {
                escribir_memoria_distribuida(ctx, offset_actual, seg->tamanio, temp_buffer);
                free(temp_buffer);
            }
            seg->base = offset_actual;
        }
        offset_actual += seg->tamanio;
    }
    
    list_clean_and_destroy_elements(ctx->huecos_libres, free);
    if (offset_actual < ctx->memoria_total) {
        t_hueco_libre* hueco_final = malloc(sizeof(t_hueco_libre));
        hueco_final->base = offset_actual;
        hueco_final->limite = ctx->memoria_total;
        list_add(ctx->huecos_libres, hueco_final);
    }
    
    log_info(ctx->logger, "## Fin de compactación");
}

void* leer_memoria_distribuida(t_memory_context* ctx, uint32_t dir_fisica, uint32_t tamanio) {
    void* buffer = malloc(tamanio);
    uint32_t leido = 0;
    
    while (leido < tamanio) {
        uint32_t dir_actual = dir_fisica + leido;
        t_memory_stick_info* stick_target = NULL;
        
        for (int i = 0; i < list_size(ctx->memory_sticks); i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            if (dir_actual >= stick->base_address && dir_actual < stick->base_address + stick->tamanio) {
                stick_target = stick;
                break;
            }
        }
        
        if (stick_target == NULL) {
            log_error(ctx->logger, "Error: Direccion fisica fuera de los limites de la memoria distribuida");
            break;
        }
        
        uint32_t offset_en_stick = dir_actual - stick_target->base_address;
        uint32_t bytes_disponibles = stick_target->tamanio - offset_en_stick;
        uint32_t bytes_a_leer = (tamanio - leido < bytes_disponibles) ? (tamanio - leido) : bytes_disponibles;
        
        enviar_peticion_lectura(offset_en_stick, bytes_a_leer, stick_target->socket_fd);
        
        int r_op = recibir_operacion(stick_target->socket_fd);
        if (r_op == LECTURA) {
            int size_buf;
            void* res = recibir_buffer(&size_buf, stick_target->socket_fd);
            memcpy(buffer + leido, res, bytes_a_leer);
            free(res);
        } else {
            log_error(ctx->logger, "Error al leer del memory stick");
            break;
        }
        
        leido += bytes_a_leer;
    }
    
    return buffer;
}

void escribir_memoria_distribuida(t_memory_context* ctx, uint32_t dir_fisica, uint32_t tamanio, void* buffer) {
    uint32_t escrito = 0;
    
    while (escrito < tamanio) {
        uint32_t dir_actual = dir_fisica + escrito;
        t_memory_stick_info* stick_target = NULL;
        
        for (int i = 0; i < list_size(ctx->memory_sticks); i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            if (dir_actual >= stick->base_address && dir_actual < stick->base_address + stick->tamanio) {
                stick_target = stick;
                break;
            }
        }
        
        if (stick_target == NULL) {
            log_error(ctx->logger, "Error: Direccion fisica fuera de los limites de la memoria distribuida");
            break;
        }
        
        uint32_t offset_en_stick = dir_actual - stick_target->base_address;
        uint32_t bytes_disponibles = stick_target->tamanio - offset_en_stick;
        uint32_t bytes_a_escribir = (tamanio - escrito < bytes_disponibles) ? (tamanio - escrito) : bytes_disponibles;
        
        enviar_peticion_escritura(offset_en_stick, bytes_a_escribir, buffer + escrito, stick_target->socket_fd);
        
        int r_op = recibir_operacion(stick_target->socket_fd);
        if (r_op == MENSAJE) {
            int size_buf;
            void* res = recibir_buffer(&size_buf, stick_target->socket_fd);
            free(res); // Esperamos MSG_OK
        } else {
            log_error(ctx->logger, "Error al escribir en memory stick");
            break;
        }
        
        escrito += bytes_a_escribir;
    }
}

    pthread_mutex_destroy(&ctx->mutex_programas);
    
    if (ctx->espacio_memoria) free(ctx->espacio_memoria);
    if (ctx->allocation_strategy) free(ctx->allocation_strategy);
    if (ctx->huecos_libres) list_destroy_and_destroy_elements(ctx->huecos_libres, free);
    if (ctx->segmentos_globales) list_destroy_and_destroy_elements(ctx->segmentos_globales, free);
    pthread_mutex_destroy(&ctx->mutex_memoria);
    
    if (ctx->swap_bitmap) bitarray_destroy(ctx->swap_bitmap);
    if (ctx->bitarray_data) free(ctx->bitarray_data);
    if (ctx->swap_socket != -1) close(ctx->swap_socket);
    
    if (ctx->base_path) free(ctx->base_path);
    if (ctx->config) config_destroy(ctx->config);
    if (ctx->logger) log_destroy(ctx->logger);
    if (ctx->socket_servidor != -1) close(ctx->socket_servidor);
    free(ctx);
}

int main(int argc, char *argv[]) {
    t_memory_context* ctx = iniciar_contexto(argc, argv);
    if (ctx == NULL) {
        return EXIT_FAILURE;
    }

    if (iniciar_servidor_memoria(ctx) == -1) {
        limpiar_contexto(ctx);
        return EXIT_FAILURE;
    }

    while (1) {
        int cliente_fd = esperar_cliente(ctx->socket_servidor, ctx->logger);
        if(cliente_fd == -1) continue;

        t_args_hilo_cliente* args = malloc(sizeof(t_args_hilo_cliente));
        args->cliente_fd = cliente_fd;
        args->ctx = ctx;

        pthread_t hilo_cliente;
        pthread_create(&hilo_cliente, NULL, (void*)atender_cliente, args);
        pthread_detach(hilo_cliente); 
    }
    
    limpiar_contexto(ctx);
    return EXIT_SUCCESS;
}

void procesar_mensaje_handshake(int cliente_fd, t_memory_context* ctx) {
    char *mensaje = recibir_handshake(cliente_fd);
    
    if (string_starts_with(mensaje, "STICK|")) {
        char **partes = string_split(mensaje, "|");
        uint32_t stick_size = atoi(partes[1]);
        char* stick_ip = strdup(partes[2]);
        int stick_puerto = atoi(partes[3]);
        
        pthread_mutex_lock(&ctx->mutex_memoria);
        
        t_memory_stick_info* stick = malloc(sizeof(t_memory_stick_info));
        stick->socket_fd = cliente_fd;
        stick->tamanio = stick_size;
        stick->base_address = ctx->memoria_total;
        stick->id = list_size(ctx->memory_sticks) + 1;
        stick->ip = stick_ip;
        stick->puerto = stick_puerto;
        
        list_add(ctx->memory_sticks, stick);
        
        t_hueco_libre* nuevo_hueco = malloc(sizeof(t_hueco_libre));
        nuevo_hueco->base = ctx->memoria_total;
        nuevo_hueco->limite = ctx->memoria_total + stick_size;
        list_add(ctx->huecos_libres, nuevo_hueco);
        
        ctx->memoria_total += stick_size;
        log_info(ctx->logger, "## Memory Stick de %d bytes Conectada en %s:%d. Memoria total: %d", stick_size, stick_ip, stick_puerto, ctx->memoria_total);
        
        pthread_mutex_unlock(&ctx->mutex_memoria);
        
        string_array_destroy(partes);
    }
    else if (string_equals_ignore_case(mensaje, HANDSHAKE_SCHEDULER)) {
        log_info(ctx->logger, "## Kernel Scheduler Conectado - FD del socket: %d", cliente_fd);
    }
    else if (string_starts_with(mensaje, HANDSHAKE_CPU)) {
        log_info(ctx->logger, "## CPU %s Conectada", mensaje);
        
        // Enviar tabla de memory sticks a la CPU
        pthread_mutex_lock(&ctx->mutex_memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = ENVIO_TABLA_STICKS; // Requiere opcode nuevo, o enviarlo como mensaje. Mejor opcode
        int cant = list_size(ctx->memory_sticks);
        agregar_a_paquete(paq, &cant, sizeof(int));
        for (int i = 0; i < cant; i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            agregar_a_paquete(paq, &(stick->id), sizeof(int));
            agregar_a_paquete(paq, &(stick->tamanio), sizeof(uint32_t));
            agregar_a_paquete(paq, &(stick->base_address), sizeof(uint32_t));
            agregar_a_paquete(paq, &(stick->puerto), sizeof(int));
            int ip_len = strlen(stick->ip) + 1;
            agregar_a_paquete(paq, &ip_len, sizeof(int));
            agregar_a_paquete(paq, stick->ip, ip_len);
        }
        enviar_paquete(paq, cliente_fd);
        eliminar_paquete(paq);
        pthread_mutex_unlock(&ctx->mutex_memoria);
    }
    else if (string_starts_with(mensaje, "SWAP|")) {
        char **partes = string_split(mensaje, "|");
        ctx->swap_file_size = atoi(partes[1]);
        ctx->swap_block_size = atoi(partes[2]);
        ctx->swap_socket = cliente_fd;
        
        int cant_bloques = ctx->swap_file_size / ctx->swap_block_size;
        int bits_bytes = (cant_bloques + 7) / 8;
        ctx->bitarray_data = malloc(bits_bytes);
        memset(ctx->bitarray_data, 0, bits_bytes);
        ctx->swap_bitmap = bitarray_create_with_mode(ctx->bitarray_data, bits_bytes, LSB_FIRST);
        
        log_info(ctx->logger, "## SWAP Conectado. Tamaño: %d, Tamaño Bloque: %d, Total Bloques: %d", ctx->swap_file_size, ctx->swap_block_size, cant_bloques);
        string_array_destroy(partes);
    }
    else { 
        log_info(ctx->logger, "## Módulo Conectado: %s", mensaje);
    }
    free(mensaje);
}

void procesar_crear_proceso_handler(int cliente_fd, t_memory_context* ctx) {
    // El sender (enviar_peticion_crear_proceso) usa buffer_write_uint32 + buffer_write_string
    // que escriben directamente al stream, NO con agregar_a_paquete.
    // Por eso debemos deserializar con buffer_reader, no con recibir_paquete.
    int size;
    void* buffer = recibir_buffer(&size, cliente_fd);
    t_buffer_reader* reader = buffer_reader_create(buffer, size);
    
    uint32_t pid = buffer_read_uint32(reader);
    char* path = buffer_read_string(reader);
    
    buffer_reader_destroy(reader);
    free(buffer);
    
    char *path_completo = string_new();
    string_append(&path_completo, ctx->base_path);
    string_append(&path_completo, "/");
    string_append(&path_completo, path);
    
    log_info(ctx->logger, "Creando proceso PID %d con script: %s", pid, path_completo);
    crear_proceso(ctx, pid, path_completo);
    enviar_mensaje(MSG_OK, cliente_fd);
    
    free(path_completo);
    free(path);
}

    }
}

void procesar_actualizacion_contexto_handler(int cliente_fd, t_memory_context* ctx) {
    int size;
    void* buffer = recibir_buffer(&size, cliente_fd);
    t_contexto_ejecucion contexto_actualizado;
    deserializar_contexto(buffer, size, &contexto_actualizado);
    free(buffer);
    
    t_programa* prog = buscar_programa(ctx, contexto_actualizado.pid);
    if (prog != NULL) {
        pthread_mutex_lock(&ctx->mutex_programas);
        
        // Actualizamos solamente los registros (que incluyen el PC)
        // La memoria es la dueña de la tabla de segmentos, no debemos sobreescribirla
        // con una version potencialmente vieja del CPU.
        prog->contexto->registros = contexto_actualizado.registros;
        
        if (contexto_actualizado.tabla_segmentos) {
            list_destroy_and_destroy_elements(contexto_actualizado.tabla_segmentos, free);
        }
        
        pthread_mutex_unlock(&ctx->mutex_programas);
        log_info(ctx->logger, "Contexto actualizado correctamente para PID %d", prog->contexto->pid);
    } else {
        log_error(ctx->logger, "Error: Intento de actualizar contexto de PID %d inexistente", contexto_actualizado.pid);
    }
}

void procesar_suspender_proceso(int cliente_fd, t_memory_context* ctx) {
    uint32_t pid;
    recv(cliente_fd, &pid, sizeof(uint32_t), MSG_WAITALL);
    
    t_programa* prog = buscar_programa(ctx, pid);
    if (!prog) {
        enviar_mensaje("ERROR", cliente_fd);
        return;
    }
    t_contexto_ejecucion* proc = prog->contexto;
    
    pthread_mutex_lock(&ctx->mutex_memoria);
    
    for (int i = 0; i < list_size(proc->tabla_segmentos); i++) {
        t_segmento* seg = list_get(proc->tabla_segmentos, i);
        if (seg->id == 0 || seg->tamanio == 0) continue; 
        
        int cant_bloques_necesarios = (seg->tamanio + ctx->swap_block_size - 1) / ctx->swap_block_size;
        int bloque_inicio = -1;
        
        int cant_bloques_swap = ctx->swap_file_size / ctx->swap_block_size;
        for (int b = 0; b <= cant_bloques_swap - cant_bloques_necesarios; b++) {
            bool libre = true;
            for (int j = 0; j < cant_bloques_necesarios; j++) {
                if (bitarray_test_bit(ctx->swap_bitmap, b + j)) {
                    libre = false;
                    break;
                }
            }
            if (libre) {
                bloque_inicio = b;
                break;
            }
        }
        
        if (bloque_inicio == -1) {
            log_error(ctx->logger, "No hay espacio en SWAP para el segmento %d del PID %d", seg->id, pid);
            continue;
        }
        
        for (int j = 0; j < cant_bloques_necesarios; j++) {
            bitarray_set_bit(ctx->swap_bitmap, bloque_inicio + j);
        }
        
        seg->bloque_swap = bloque_inicio;
        
        void* buffer_seg = leer_memoria_distribuida(ctx, seg->base, seg->tamanio);
        
        int offset = 0;
        for (int j = 0; j < cant_bloques_necesarios; j++) {
            void* bloque_data = malloc(ctx->swap_block_size);
            memset(bloque_data, 0, ctx->swap_block_size);
            int copy_size = (offset + ctx->swap_block_size <= seg->tamanio) ? ctx->swap_block_size : (seg->tamanio - offset);
            memcpy(bloque_data, buffer_seg + offset, copy_size);
            
            enviar_int(ctx->swap_socket, ESCRITURA);
            enviar_int(ctx->swap_socket, bloque_inicio + j);
            enviar_int(ctx->swap_socket, ctx->swap_block_size);
            send(ctx->swap_socket, bloque_data, ctx->swap_block_size, 0);
            
            recibir_mensaje(ctx->swap_socket, ctx->logger);
            free(bloque_data);
            offset += copy_size;
        }
        free(buffer_seg);
        
        // Liberar espacio en memoria principal
        for (int h = 0; h < list_size(ctx->segmentos_globales); h++) {
            t_segmento* gs = list_get(ctx->segmentos_globales, h);
            if (gs->base == seg->base && gs->id == seg->id) {
                t_hueco_libre* hueco = malloc(sizeof(t_hueco_libre));
                hueco->base = seg->base;
                hueco->limite = seg->base + seg->tamanio;
                list_add(ctx->huecos_libres, hueco);
                
                list_remove_and_destroy_element(ctx->segmentos_globales, h, free);
                break;
            }
        }
    }
    
    // Consolidar huecos si es posible
    // (Omitido para brevedad o se puede llamar a una funcion de consolidar)
    
    pthread_mutex_unlock(&ctx->mutex_memoria);
    
    enviar_mensaje("OK", cliente_fd);
    log_info(ctx->logger, "Proceso %d suspendido y guardado en SWAP", pid);
}

void procesar_desuspender_proceso(int cliente_fd, t_memory_context* ctx) {
    uint32_t pid;
    recv(cliente_fd, &pid, sizeof(uint32_t), MSG_WAITALL);
    
    t_programa* prog = buscar_programa(ctx, pid);
    if (!prog) {
        enviar_mensaje("ERROR", cliente_fd);
        return;
    }
    t_contexto_ejecucion* proc = prog->contexto;
    
    pthread_mutex_lock(&ctx->mutex_memoria);
    
    bool hay_espacio = true;
    for (int i = 0; i < list_size(proc->tabla_segmentos); i++) {
        t_segmento* seg = list_get(proc->tabla_segmentos, i);
        if (seg->id == 0 || seg->tamanio == 0) continue;
        
        // Buscar hueco en memoria principal (First Fit para simplificar)
        t_hueco_libre* best_hueco = NULL;
        for (int j = 0; j < list_size(ctx->huecos_libres); j++) {
            t_hueco_libre* h = list_get(ctx->huecos_libres, j);
            if ((h->limite - h->base) >= seg->tamanio) {
                best_hueco = h;
                break;
            }
        }
        if (!best_hueco) {
            hay_espacio = false;
            break;
        }
    }
    
    if (!hay_espacio) {
        pthread_mutex_unlock(&ctx->mutex_memoria);
        enviar_mensaje("ERROR", cliente_fd);
        log_error(ctx->logger, "No hay espacio en Memoria Principal para desuspender PID %d", pid);
        return;
    }
    
    for (int i = 0; i < list_size(proc->tabla_segmentos); i++) {
        t_segmento* seg = list_get(proc->tabla_segmentos, i);
        if (seg->id == 0 || seg->tamanio == 0) continue;
        
        t_hueco_libre* best_hueco = NULL;
        for (int j = 0; j < list_size(ctx->huecos_libres); j++) {
            t_hueco_libre* h = list_get(ctx->huecos_libres, j);
            if ((h->limite - h->base) >= seg->tamanio) {
                best_hueco = h;
                break;
            }
        }
        
        // Asignar base
        seg->base = best_hueco->base;
        best_hueco->base += seg->tamanio;
        if (best_hueco->base == best_hueco->limite) {
            list_remove_element(ctx->huecos_libres, best_hueco);
            free(best_hueco);
        }
        
        t_segmento* nuevo_seg = malloc(sizeof(t_segmento));
        nuevo_seg->id = seg->id;
        nuevo_seg->tamanio = seg->tamanio;
        nuevo_seg->base = seg->base;
        nuevo_seg->bloque_swap = -1;
        list_add(ctx->segmentos_globales, nuevo_seg);
        
        // Leer de SWAP y escribir en Memoria Principal
        int cant_bloques = (seg->tamanio + ctx->swap_block_size - 1) / ctx->swap_block_size;
        int offset = 0;
        void* buffer_seg = malloc(seg->tamanio);
        
        for (int j = 0; j < cant_bloques; j++) {
            enviar_int(ctx->swap_socket, LECTURA);
            enviar_int(ctx->swap_socket, seg->bloque_swap + j);
            
            int opcode = recibir_operacion(ctx->swap_socket);
            if (opcode == LECTURA) {
                int size_buf;
                void* res = recibir_buffer(&size_buf, ctx->swap_socket);
                int copy_size = (offset + ctx->swap_block_size <= seg->tamanio) ? ctx->swap_block_size : (seg->tamanio - offset);
                memcpy(buffer_seg + offset, res, copy_size);
                free(res);
            }
            offset += ctx->swap_block_size;
            bitarray_clean_bit(ctx->swap_bitmap, seg->bloque_swap + j);
        }
        
        escribir_memoria_distribuida(ctx, seg->base, seg->tamanio, buffer_seg);
        free(buffer_seg);
        seg->bloque_swap = -1;
    }
    
    pthread_mutex_unlock(&ctx->mutex_memoria);
    enviar_mensaje("OK", cliente_fd);
    log_info(ctx->logger, "Proceso %d desuspendido y traído a Memoria Principal", pid);
}

void atender_cliente(void *arg) {
    t_args_hilo_cliente* args = (t_args_hilo_cliente*)arg;
    int cliente_fd = args->cliente_fd;
    t_memory_context* ctx = args->ctx;
    free(args);

    while (1) {
        int cod_op = recibir_operacion(cliente_fd);

        switch(cod_op) {
            case MENSAJE:
                procesar_mensaje_handshake(cliente_fd, ctx);
                break;
            case CREAR_PROCESO:
                procesar_crear_proceso_handler(cliente_fd, ctx);
                break;
            case SOLICITUD_CONTEXTO:
                procesar_solicitud_contexto_handler(cliente_fd, ctx);
                break;
            case ACTUALIZACION_CONTEXTO:
                procesar_actualizacion_contexto_handler(cliente_fd, ctx);
                break;
            case SUSPENDER_PROCESO:
                procesar_suspender_proceso(cliente_fd, ctx);
                break;
            case DESUSPENDER_PROCESO:
                procesar_desuspender_proceso(cliente_fd, ctx);
                break;
            case FETCH_INSTRUCCION:
                procesar_fetch_instruccion_handler(cliente_fd, ctx);
                break;
            case CREAR_SEGMENTO:
            {
                t_list* paquete = recibir_paquete(cliente_fd);
                uint32_t pid = *(uint32_t*)list_get(paquete, 0);
                uint32_t id_seg = *(uint32_t*)list_get(paquete, 1);
                uint32_t tamanio_seg = *(uint32_t*)list_get(paquete, 2);
                list_destroy_and_destroy_elements(paquete, free);

                int needs_compaction = 0;
                pthread_mutex_lock(&ctx->mutex_memoria);
                t_segmento* nuevo_seg = crear_segmento(ctx, id_seg, tamanio_seg, &needs_compaction);
                
                if (needs_compaction) {
                    t_paquete* paq = crear_paquete();
                    paq->codigo_operacion = SOLICITUD_COMPACTACION;
                    enviar_paquete(paq, cliente_fd);
                    eliminar_paquete(paq);
                    
                    int res = recibir_operacion(cliente_fd);
                    if (res == RESPUESTA_COMPACTACION) {
                        compactar_memoria(ctx);
                        nuevo_seg = crear_segmento(ctx, id_seg, tamanio_seg, &needs_compaction);
                    }
                }

                if (nuevo_seg != NULL) {
                    t_programa* prog = buscar_programa(ctx, pid);
                    if (prog != NULL) {
                        pthread_mutex_lock(&ctx->mutex_programas);
                        if (prog->contexto->tabla_segmentos == NULL) {
                            prog->contexto->tabla_segmentos = list_create();
                        }
                        list_add(prog->contexto->tabla_segmentos, nuevo_seg);
                        log_info(ctx->logger, "DEBUG: Added segment to PID %d. Total segments now: %d", pid, list_size(prog->contexto->tabla_segmentos));
                        pthread_mutex_unlock(&ctx->mutex_programas);
                    } else {
                        log_error(ctx->logger, "DEBUG: prog was NULL for PID %d in CREAR_SEGMENTO!", pid);
                    }
                    enviar_mensaje(MSG_OK, cliente_fd);
                } else {
                    enviar_mensaje("OUT_OF_MEMORY", cliente_fd);
                }
                pthread_mutex_unlock(&ctx->mutex_memoria);
                break;
            }
            case LIBERAR_PROCESO:
            {
                t_list* paquete = recibir_paquete(cliente_fd);
                uint32_t pid_a_liberar = *(uint32_t*)list_get(paquete, 0);
                list_destroy_and_destroy_elements(paquete, free);

                pthread_mutex_lock(&ctx->mutex_memoria);
                t_programa* prog = buscar_programa(ctx, pid_a_liberar);
                if (prog != NULL && prog->contexto->tabla_segmentos != NULL) {
                    pthread_mutex_lock(&ctx->mutex_programas);
                    for (int i = 0; i < list_size(prog->contexto->tabla_segmentos); i++) {
                        t_segmento* seg = list_get(prog->contexto->tabla_segmentos, i);
                        eliminar_segmento(ctx, seg);
                    }
                    list_clean(prog->contexto->tabla_segmentos);
                    pthread_mutex_unlock(&ctx->mutex_programas);
                }
                log_info(ctx->logger, "Proceso %d liberado de la memoria", pid_a_liberar);
                pthread_mutex_unlock(&ctx->mutex_memoria);
                break;
            }
            case ELIMINAR_SEGMENTO:
            {
                t_list* paquete = recibir_paquete(cliente_fd);
                uint32_t pid = *(uint32_t*)list_get(paquete, 0);
                uint32_t id_seg = *(uint32_t*)list_get(paquete, 1);
                list_destroy_and_destroy_elements(paquete, free);

                pthread_mutex_lock(&ctx->mutex_memoria);
                t_programa* prog = buscar_programa(ctx, pid);
                if (prog != NULL && prog->contexto->tabla_segmentos != NULL) {
                    pthread_mutex_lock(&ctx->mutex_programas);
                    t_segmento* seg_a_eliminar = NULL;
                    for (int i = 0; i < list_size(prog->contexto->tabla_segmentos); i++) {
                        t_segmento* seg = list_get(prog->contexto->tabla_segmentos, i);
                        if (seg->id == id_seg) {
                            seg_a_eliminar = seg;
                            list_remove(prog->contexto->tabla_segmentos, i);
                            break;
                        }
                    }
                    pthread_mutex_unlock(&ctx->mutex_programas);

                    if (seg_a_eliminar != NULL) {
                        eliminar_segmento(ctx, seg_a_eliminar);
                        enviar_mensaje(MSG_OK, cliente_fd);
                    } else {
                        enviar_mensaje("SEG_NOT_FOUND", cliente_fd);
                    }
                } else {
                    enviar_mensaje("PROG_NOT_FOUND", cliente_fd);
                }
                pthread_mutex_unlock(&ctx->mutex_memoria);
                break;
            }
            case SOLICITUD_TAM_MAX_SEGMENTO:
            {
                t_paquete* paq = crear_paquete();
                paq->codigo_operacion = TAM_MAX_SEGMENTO;
                agregar_a_paquete(paq, &ctx->segment_max_size, sizeof(int));
                enviar_paquete(paq, cliente_fd);
                eliminar_paquete(paq);
                break;
            }
            case LEER_MEMORIA:
            {
                int size;
                void* buffer = recibir_buffer(&size, cliente_fd);
                uint32_t dir_fisica, tamanio;
                int desp = 0;
                memcpy(&dir_fisica, buffer + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                memcpy(&tamanio, buffer + desp, sizeof(uint32_t));
                free(buffer);

                pthread_mutex_lock(&ctx->mutex_memoria);
                void* data_leida = leer_memoria_distribuida(ctx, dir_fisica, tamanio);
                pthread_mutex_unlock(&ctx->mutex_memoria);

                t_paquete* paq = crear_paquete();
                paq->codigo_operacion = MENSAJE; 
                agregar_a_paquete(paq, data_leida, tamanio);
                enviar_paquete(paq, cliente_fd);
                eliminar_paquete(paq);
                free(data_leida);
                
                break;
            }
            case ESCRIBIR_MEMORIA:
            {
                int size;
                void* buffer = recibir_buffer(&size, cliente_fd);
                uint32_t dir_fisica, tamanio;
                int desp = 0;
                memcpy(&dir_fisica, buffer + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                memcpy(&tamanio, buffer + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                
                int tamanio_rep;
                memcpy(&tamanio_rep, buffer + desp, sizeof(int)); desp += sizeof(int);
                
                pthread_mutex_lock(&ctx->mutex_memoria);
                escribir_memoria_distribuida(ctx, dir_fisica, tamanio, buffer + desp);
                pthread_mutex_unlock(&ctx->mutex_memoria);
                
                free(buffer);
                enviar_mensaje(MSG_OK, cliente_fd);
                break;
            }
            case CONSULTAR_ESPACIO_LIBRE:
            {
                int espacio_fijo_mock = 999999; 
                send(cliente_fd, &espacio_fijo_mock, sizeof(int), 0); 
                break;
            }
            case -1:
                log_warning(ctx->logger, "Cliente desconectado");
                close(cliente_fd);
                return;
            default:
                log_warning(ctx->logger, "Operación desconocida: %d", cod_op);
                close(cliente_fd);
                return;
        }
    }
}

        enviar_paquete(paquete_pid, socket_cpu);
        eliminar_paquete(paquete_pid);
        
        log_info(ctx->logger, "Se envió el PID %d a la CPU para ejecutar", proceso_a_ejecutar->pid);
        planificador_iniciar_quantum_si_corresponde(ctx->estratega, proceso_a_ejecutar, socket_cpu);

    paquete_pid->codigo_operacion = PID_A_EJECUTAR;

void procesar_mensaje_handshake(int cliente_fd, t_scheduler_context* ctx) {
    int size;
    char* mensaje = recibir_buffer(&size, cliente_fd);

    if (string_starts_with(mensaje, HANDSHAKE_CPU)) {
        log_info(ctx->logger, "## CPU %s Conectada", mensaje);
        ctx->estratega->socket_cpu = cliente_fd;
        despachar_proceso_a_cpu(ctx, cliente_fd);
    } else {
        t_interfaz_io* interfaz = buscar_interfaz_por_socket(ctx->estratega, cliente_fd);
        if (interfaz != NULL) {
            // Mensaje de una IO ya registrada
            if (string_equals_ignore_case(mensaje, "OK")) {
                planificador_io_terminada(ctx->estratega, cliente_fd);
                if (ctx->estratega->proceso_exec == NULL && ctx->estratega->socket_cpu != -1) {
                    despachar_proceso_a_cpu(ctx, ctx->estratega->socket_cpu);
                }
            } else {
                // Es el texto leido por STDIN
                t_proceso_planificable* proceso = NULL;
                if (interfaz->solicitud_actual != NULL) {
                    proceso = interfaz->solicitud_actual->proceso;
                }
                
                if (proceso != NULL) {
                    t_paquete* paq_write = crear_paquete();
                    paq_write->codigo_operacion = ESCRIBIR_MEMORIA;
                    agregar_a_paquete(paq_write, &(proceso->io_dir_fisica), sizeof(uint32_t));
                    agregar_a_paquete(paq_write, &(proceso->io_tamanio), sizeof(uint32_t));
                    agregar_a_paquete(paq_write, mensaje, proceso->io_tamanio);
                    enviar_paquete(paq_write, ctx->socket_memoria);
                    eliminar_paquete(paq_write);
                    
                    int r_mem = recibir_operacion(ctx->socket_memoria);
                    if (r_mem == MENSAJE) {
                        int s_m; char* msj_m = recibir_buffer(&s_m, ctx->socket_memoria); free(msj_m);
                    }
                }
                
                planificador_io_terminada(ctx->estratega, cliente_fd);
                if (ctx->estratega->proceso_exec == NULL && ctx->estratega->socket_cpu != -1) {
                    despachar_proceso_a_cpu(ctx, ctx->estratega->socket_cpu);
                }
            }
        } else {
            // Es el nombre de una nueva interfaz registrandose
            planificador_registrar_io(ctx->estratega, mensaje, cliente_fd);
        }
    }
    free(mensaje);
}

static void manejar_io_sleep(t_scheduler_context* ctx, t_list* paquete, uint32_t pid) {
    uint32_t* tiempo_ptr = list_get(paquete, 1);

        int op = PETICION_IO;
        send(interfaz->socket_fd, &op, sizeof(int), 0);
        send(interfaz->socket_fd, &pid, sizeof(uint32_t), 0);
        send(interfaz->socket_fd, tiempo_ptr, sizeof(uint32_t), 0);
    }
    planificador_atender_peticion_io(ctx->estratega, ctx->estratega->proceso_exec, "SLEEP");
}

static void manejar_io_stdout(t_scheduler_context* ctx, t_list* paquete, uint32_t pid) {
    uint32_t* dir_fisica_ptr = list_get(paquete, 1);
    uint32_t* tamanio_ptr = list_get(paquete, 2);
    uint32_t dir_fisica = *dir_fisica_ptr;
    uint32_t tamanio = *tamanio_ptr;
    
    t_interfaz_io* interfaz = buscar_interfaz_por_nombre(ctx->estratega, "STDOUT");
    if (interfaz != NULL) {
        // Pedimos leer a memoria
        t_paquete* paq_mem = crear_paquete();
        paq_mem->codigo_operacion = LEER_MEMORIA;
        agregar_a_paquete(paq_mem, &dir_fisica, sizeof(uint32_t));
        agregar_a_paquete(paq_mem, &tamanio, sizeof(uint32_t));
        enviar_paquete(paq_mem, ctx->socket_memoria);
        eliminar_paquete(paq_mem);
        
        int op = recibir_operacion(ctx->socket_memoria);
        if (op == MENSAJE) {
            int size;
            void* buf = recibir_buffer(&size, ctx->socket_memoria);
            
            int op_io = PETICION_IO;
            send(interfaz->socket_fd, &op_io, sizeof(int), 0);
            send(interfaz->socket_fd, &pid, sizeof(uint32_t), 0);
            
            // Mandamos el tamaño y el buffer al IO
            int len = tamanio + 1;
            char* dummy = malloc(len);
            memcpy(dummy, buf, tamanio);
            dummy[tamanio] = '\0';
            
            send(interfaz->socket_fd, &len, sizeof(int), 0);
            send(interfaz->socket_fd, dummy, len, 0);
            free(dummy);
            free(buf);
        }
    }
    planificador_atender_peticion_io(ctx->estratega, ctx->estratega->proceso_exec, "STDOUT");
}

void procesar_syscall_io(int cliente_fd, t_scheduler_context* ctx, int tipo_syscall) {
    t_list* paquete = recibir_paquete(cliente_fd);
    uint32_t* pid_ptr = list_get(paquete, 0);
    uint32_t pid = *pid_ptr;
    
    // Guard Clause: Validación del proceso actual
    if (ctx->estratega->proceso_exec == NULL || ctx->estratega->proceso_exec->pid != pid) {
        despachar_proceso_a_cpu(ctx, cliente_fd);
        list_destroy_and_destroy_elements(paquete, free);
        return;
    }

    switch (tipo_syscall) {
        case SYSCALL_SLEEP:
            manejar_io_sleep(ctx, paquete, pid);
            break;
        case SYSCALL_STDOUT:
            manejar_io_stdout(ctx, paquete, pid);
            break;
        case SYSCALL_STDIN:
            manejar_io_stdin(ctx, paquete, pid);
            break;
    }
    
    despachar_proceso_a_cpu(ctx, cliente_fd);
    list_destroy_and_destroy_elements(paquete, free);
}

void procesar_syscall_exit(int cliente_fd, t_scheduler_context* ctx) {
    t_list* paquete = recibir_paquete(cliente_fd);
    if (ctx->estratega->proceso_exec != NULL) {
        log_info(ctx->logger, "## Finaliza el proceso %d - Motivo: SUCCESS", ctx->estratega->proceso_exec->pid);
        ctx->estratega->proceso_exec->estado = ESTADO_PROCESO_EXIT;
        
        t_paquete* paq_liberar = crear_paquete();
        paq_liberar->codigo_operacion = LIBERAR_PROCESO;
        agregar_a_paquete(paq_liberar, &(ctx->estratega->proceso_exec->pid), sizeof(uint32_t));
        enviar_paquete(paq_liberar, ctx->socket_memoria);
        eliminar_paquete(paq_liberar);
        
        planificador_liberar_exec(ctx->estratega, ctx->estratega->proceso_exec->pid);
    }
    despachar_proceso_a_cpu(ctx, cliente_fd);
    list_destroy_and_destroy_elements(paquete, free);
}

void procesar_syscall_mutex(int cliente_fd, t_scheduler_context* ctx, int tipo_syscall) {
    t_list* paquete = recibir_paquete(cliente_fd);
    uint32_t* pid_ptr = list_get(paquete, 0);
    uint32_t pid = *pid_ptr;
    char* recurso = list_get(paquete, 1);

    // Guard Clause
    if (ctx->estratega->proceso_exec == NULL || ctx->estratega->proceso_exec->pid != pid) {
        despachar_proceso_a_cpu(ctx, cliente_fd);
        list_destroy_and_destroy_elements(paquete, free);
        return;
    }

    switch (tipo_syscall) {
        case SYSCALL_MUTEX_CREATE:
            planificador_mutex_create(ctx->estratega, ctx->estratega->proceso_exec, recurso);
            devolver_ejecucion_a_cpu(ctx, cliente_fd);
            break;

        case SYSCALL_MUTEX_LOCK: {
            int se_bloqueo = planificador_mutex_lock(ctx->estratega, ctx->estratega->proceso_exec, recurso);
            if (se_bloqueo == 0) {
                devolver_ejecucion_a_cpu(ctx, cliente_fd); // Sigue en EXEC
            } else {
                despachar_proceso_a_cpu(ctx, cliente_fd); // Se bloqueó
            }
            break;
        }

        case SYSCALL_MUTEX_UNLOCK:
            planificador_mutex_unlock(ctx->estratega, ctx->estratega->proceso_exec, recurso);
            devolver_ejecucion_a_cpu(ctx, cliente_fd); // Sigue en EXEC
            break;
    }
    
    list_destroy_and_destroy_elements(paquete, free);
}

void procesar_syscall_memoria(int cliente_fd, t_scheduler_context* ctx, int tipo_syscall) {
    t_list* paquete = recibir_paquete(cliente_fd);
    uint32_t* pid_ptr = list_get(paquete, 0);
    uint32_t pid = *pid_ptr;

    // Guard Clause: Validación del proceso actual
    if (ctx->estratega->proceso_exec == NULL || ctx->estratega->proceso_exec->pid != pid) {
        despachar_proceso_a_cpu(ctx, cliente_fd);
        list_destroy_and_destroy_elements(paquete, free);
        return;
    }

    if (tipo_syscall == SYSCALL_MEM_ALLOC) {
        uint32_t* id_seg_ptr = list_get(paquete, 1);
        uint32_t* tamanio_ptr = list_get(paquete, 2);
        
        t_paquete* paq_mem = crear_paquete();
        paq_mem->codigo_operacion = CREAR_SEGMENTO;
        agregar_a_paquete(paq_mem, &pid, sizeof(uint32_t));
        agregar_a_paquete(paq_mem, id_seg_ptr, sizeof(uint32_t));
        agregar_a_paquete(paq_mem, tamanio_ptr, sizeof(uint32_t));
        enviar_paquete(paq_mem, ctx->socket_memoria);
        eliminar_paquete(paq_mem);

        int respuesta = recibir_operacion(ctx->socket_memoria);
        
        if (respuesta == SOLICITUD_COMPACTACION) {
            log_info(ctx->logger, "Compactación requerida. Avisando a Memoria que puede proceder...");
            t_paquete* paq_resp = crear_paquete();
            paq_resp->codigo_operacion = RESPUESTA_COMPACTACION;
            enviar_paquete(paq_resp, ctx->socket_memoria);
            eliminar_paquete(paq_resp);
            
            // Esperamos la respuesta final de la creación del segmento post-compactación
            respuesta = recibir_operacion(ctx->socket_memoria);
        }

        if (respuesta == MENSAJE) {
            int size;
            char* msj = recibir_buffer(&size, ctx->socket_memoria);
            if (string_equals_ignore_case(msj, "OK")) {
                devolver_ejecucion_a_cpu(ctx, cliente_fd);
            } else {
                log_error(ctx->logger, "Out of memory. Finalizando proceso %d", pid);
                ctx->estratega->proceso_exec->estado = ESTADO_PROCESO_EXIT;
                
                t_paquete* paq_liberar = crear_paquete();
                paq_liberar->codigo_operacion = LIBERAR_PROCESO;
                agregar_a_paquete(paq_liberar, &pid, sizeof(uint32_t));
                enviar_paquete(paq_liberar, ctx->socket_memoria);
                eliminar_paquete(paq_liberar);

                planificador_liberar_exec(ctx->estratega, pid);
                despachar_proceso_a_cpu(ctx, cliente_fd);
            }
            free(msj);
        }
    } else if (tipo_syscall == SYSCALL_MEM_FREE) {
        uint32_t* id_seg_ptr = list_get(paquete, 1);

        t_paquete* paq_mem = crear_paquete();
        paq_mem->codigo_operacion = ELIMINAR_SEGMENTO;
        agregar_a_paquete(paq_mem, &pid, sizeof(uint32_t));
        agregar_a_paquete(paq_mem, id_seg_ptr, sizeof(uint32_t));
        enviar_paquete(paq_mem, ctx->socket_memoria);
        eliminar_paquete(paq_mem);

        int respuesta = recibir_operacion(ctx->socket_memoria);
        if (respuesta == MENSAJE) {
            int size;
            char* msj = recibir_buffer(&size, ctx->socket_memoria);
            if (string_equals_ignore_case(msj, "OK")) {
                devolver_ejecucion_a_cpu(ctx, cliente_fd);
            }
            free(msj);
        }
    }
    
    list_destroy_and_destroy_elements(paquete, free);
}

void procesar_syscall_init_proc(int cliente_fd, t_scheduler_context* ctx) {
    t_list* paquete = recibir_paquete(cliente_fd);
    uint32_t* pid_ptr = list_get(paquete, 0);
    uint32_t pid = *pid_ptr;
    char* script_path = list_get(paquete, 1);
    uint32_t* prioridad_ptr = list_get(paquete, 2);
    uint32_t prioridad = *prioridad_ptr;

    // Guard Clause
    if (ctx->estratega->proceso_exec == NULL || ctx->estratega->proceso_exec->pid != pid) {
        despachar_proceso_a_cpu(ctx, cliente_fd);
        list_destroy_and_destroy_elements(paquete, free);
        return;
    }

    static int contador_pid = 1;
    int nuevo_pid = contador_pid++;

    log_info(ctx->logger, "## Proceso %d solicita crear nuevo proceso con script %s", pid, script_path);

    enviar_peticion_crear_proceso(nuevo_pid, script_path, ctx->socket_memoria);

    t_proceso_planificable* pcb_nuevo = malloc(sizeof(t_proceso_planificable));
    pcb_nuevo->pid = nuevo_pid;
    pcb_nuevo->cola_prioridad = prioridad;
    pcb_nuevo->bloqueo_id = 0;

    planificador_encolar_new(ctx->estratega, pcb_nuevo);
    planificador_admitir_a_ready(ctx->estratega);

    devolver_ejecucion_a_cpu(ctx, cliente_fd); // Sigue en EXEC

    list_destroy_and_destroy_elements(paquete, free);
}

// 2. Aislamos la lógica compleja del SEG_FAULT en su propio handler
int handler_seg_fault(int cliente_fd, t_scheduler_context* ctx) {
    t_list* paquete = recibir_paquete(cliente_fd);
    
    if (ctx->estratega->proceso_exec != NULL) {
        log_info(ctx->logger, "## Finaliza el proceso %d - Motivo: SEG_FAULT", ctx->estratega->proceso_exec->pid);
        ctx->estratega->proceso_exec->estado = ESTADO_PROCESO_EXIT;

        t_paquete* paq_liberar = crear_paquete();
        paq_liberar->codigo_operacion = LIBERAR_PROCESO;
        agregar_a_paquete(paq_liberar, &(ctx->estratega->proceso_exec->pid), sizeof(uint32_t));
        enviar_paquete(paq_liberar, ctx->socket_memoria);
        eliminar_paquete(paq_liberar);

        planificador_liberar_exec(ctx->estratega, ctx->estratega->proceso_exec->pid);
    }
    
    despachar_proceso_a_cpu(ctx, cliente_fd);
    list_destroy_and_destroy_elements(paquete, free);
    
    return 1; // Retornamos 1 para indicar que el hilo debe continuar escuchando
}

// Creamos un array estático de tamaño 256 (suficiente para cubrir todos los opcodes).
// Los índices que no declaremos se inicializarán automáticamente en NULL.
static const handler_t tabla_despacho[256] = {
    [MENSAJE]              = handler_mensaje,
    [SYSCALL_EXIT]         = handler_exit,
    [SYSCALL_SLEEP]        = handler_sleep,
    [SYSCALL_STDOUT]       = handler_stdout,
    [SYSCALL_STDIN]        = handler_stdin,
    [SYSCALL_MUTEX_CREATE] = handler_mutex_create,
    [SYSCALL_MUTEX_LOCK]   = handler_mutex_lock,
    [SYSCALL_MUTEX_UNLOCK] = handler_mutex_unlock,
    [SYSCALL_SEG_FAULT]    = handler_seg_fault,
    [SYSCALL_MEM_ALLOC]    = handler_mem_alloc,
    [SYSCALL_MEM_FREE]     = handler_mem_free,
    [SYSCALL_INIT_PROC]    = handler_init_proc,
    [FIN_DE_QUANTUM]       = handler_fin_quantum
};


//Por último, tu función orquestadora queda extremadamente simple y segura. Comprueba 
//límites para evitar un Segmentation Fault, busca la función en la tabla mediante el 
//índice y la ejecuta [cite: 450].

int atenderRespuesta(int cliente_fd, t_scheduler_context* ctx, int mensaje_cliente) {
    // 1. Manejo de desconexión
    if (mensaje_cliente == -1) {
        log_warning(ctx->logger, "El cliente se desconectó.");
        return -1; // -1 indica que se debe cortar la conexión
    }

    // 2. Validación de seguridad (Bounds Checking) y función no mapeada
    if (mensaje_cliente < 0 || mensaje_cliente >= 256 || tabla_despacho[mensaje_cliente] == NULL) {
        log_warning(ctx->logger, "No entiendo el mensaje o código de operación: %d", mensaje_cliente);
        return -1; // -1 indica error / cortar conexión
    }

    // 3. ¡Magia! Ejecuta el handler correspondiente en O(1)
    return tabla_despacho[mensaje_cliente](cliente_fd, ctx);
}

void atender_cliente_scheduler(void* arg) {
    t_args_hilo_scheduler* args = (t_args_hilo_scheduler*)arg;
    int cliente_fd = args->cliente_fd;
    t_scheduler_context* ctx = args->ctx;
    free(args);

    while (1) {
        int mensaje_cliente = recibir_operacion(cliente_fd);
        
        // Ejecutamos la respuesta
        int status = atenderRespuesta(cliente_fd, ctx, mensaje_cliente);
        
        // Si la tabla o la validación retornan -1, rompemos el ciclo y cerramos.
        if (status == -1) {
            close(cliente_fd);
            break;
        }
    }
}

#include "colas.h"

t_algoritmo_cola algoritmo_cola_desde_string(char* algoritmo) {
    if (algoritmo == NULL) {
        return ALGORITMO_COLA_INVALIDO;
    }

    if (string_equals_ignore_case(algoritmo, "FIFO")) {
        return ALGORITMO_COLA_FIFO;
    }

    if (string_equals_ignore_case(algoritmo, "RR")) {
        return ALGORITMO_COLA_RR;
    }

    return ALGORITMO_COLA_INVALIDO;
}

t_list* crear_colas_ready_desde_config(t_config* config, int quantum_rr, t_log* logger) {
    t_list* colas = list_create();
    char** algoritmos_colas = config_get_array_value(config, "QUEUES_ALGORITHMS");

    for (int i = 0; algoritmos_colas != NULL && algoritmos_colas[i] != NULL; i++) {
        t_algoritmo_cola algoritmo = algoritmo_cola_desde_string(algoritmos_colas[i]);

        if (algoritmo == ALGORITMO_COLA_INVALIDO) {
            log_error(logger, "Algoritmo de cola invalido en nivel %d: %s", i, algoritmos_colas[i]);
            string_array_destroy(algoritmos_colas);
            list_destroy_and_destroy_elements(colas, destruir_cola_planificacion);
            return NULL;
        }

        t_cola_planificacion* cola = calloc(1, sizeof(t_cola_planificacion));
        cola->nivel = i;
        cola->algoritmo = algoritmo;
        cola->quantum = algoritmo == ALGORITMO_COLA_RR ? quantum_rr : 0;
        cola->procesos = queue_create();
        list_add(colas, cola);
    }

    if (algoritmos_colas != NULL) {
        string_array_destroy(algoritmos_colas);
    }

    return colas;
}

void destruir_cola_planificacion(void* elemento) {
    t_cola_planificacion* cola = elemento;

    if (cola == NULL) {
        return;
    }

    queue_destroy(cola->procesos);
    free(cola);
}

t_cola_planificacion* obtener_cola_planificacion(t_list* colas, int cola_prioridad) {
    if (colas == NULL || cola_prioridad < 0 || cola_prioridad >= list_size(colas)) {
        return NULL;
    }

    return list_get(colas, cola_prioridad);
}

bool encolar_en_cola_ready(t_list* colas, t_proceso_planificable* proceso) {
    t_cola_planificacion* cola = obtener_cola_planificacion(colas, proceso != NULL ? proceso->cola_prioridad : -1);

    if (cola == NULL) {
        return false;
    }

    proceso->estado = ESTADO_PROCESO_READY;
    queue_push(cola->procesos, proceso);
    return true;
}

t_proceso_planificable* desencolar_siguiente_ready(t_list* colas) {
    if (colas == NULL) {
        return NULL;
    }

    for (int i = 0; i < list_size(colas); i++) {
        t_cola_planificacion* cola = list_get(colas, i);

        if (!queue_is_empty(cola->procesos)) {
            return queue_pop(cola->procesos);
        }
    }

    return NULL;
}

int quantum_para_cola(t_list* colas, int cola_prioridad) {
    t_cola_planificacion* cola = obtener_cola_planificacion(colas, cola_prioridad);

    if (cola == NULL) {
        return 0;
    }

    return cola->quantum;
}

t_scheduler_context* iniciar_contexto(int argc, char* argv[]) {
    t_scheduler_context* ctx = malloc(sizeof(t_scheduler_context));
    
    if (argc != 3) {
        printf("Uso: ./bin/kernel_scheduler [Archivo Config] [Path Proceso Inicial]\n");
        free(ctx);
        exit(EXIT_FAILURE);
    }
    
    char* path_config = argv[1];
    ctx->path_script_inicial = argv[2];

    ctx->config = config_create(path_config);
    if (ctx->config == NULL) {
        printf("Error: No se pudo leer la configuración en %s\n", path_config);
        free(ctx);
        exit(EXIT_FAILURE);
    }

    char* log_level_str = config_get_string_value(ctx->config, "LOG_LEVEL");
    t_log_level level = LOG_LEVEL_DEBUG;
    if (log_level_str != NULL) {
        if (string_equals_ignore_case(log_level_str, "TRACE")) level = LOG_LEVEL_TRACE;
        else if (string_equals_ignore_case(log_level_str, "DEBUG")) level = LOG_LEVEL_DEBUG;
        else if (string_equals_ignore_case(log_level_str, "INFO")) level = LOG_LEVEL_INFO;
        else if (string_equals_ignore_case(log_level_str, "WARNING")) level = LOG_LEVEL_WARNING;
        else if (string_equals_ignore_case(log_level_str, "ERROR")) level = LOG_LEVEL_ERROR;
    }

    ctx->logger = log_create("Scheduler.log", "Scheduler", true, level);

    ctx->ip_memory = config_get_string_value(ctx->config, "IP_MEMORY");
    ctx->puerto_memory = config_get_string_value(ctx->config, "PUERTO_MEMORY");
    ctx->puerto_escucha = config_get_string_value(ctx->config, "PUERTO_SCHEDULER");
    ctx->socket_memoria = -1;
    ctx->socket_escucha = -1;

    // Crear el estratega (contexto de negocio)
    ctx->estratega = crear_estratega_planificacion(ctx->config, ctx->logger);

    return ctx;
}

void conectar_kernel_memory(t_scheduler_context* ctx) {
    ctx->socket_memoria = crear_conexion(ctx->ip_memory, ctx->puerto_memory);
    if(ctx->socket_memoria != -1) {
        log_info(ctx->logger, "## Conectado a Kernel Memory");
        enviar_mensaje(HANDSHAKE_SCHEDULER, ctx->socket_memoria);
        ctx->estratega->socket_memoria = ctx->socket_memoria;
    } else {
        log_error(ctx->logger, "Error al conectar con Kernel Memory en %s:%s", ctx->ip_memory, ctx->puerto_memory);
        limpiar_contexto(ctx);
        exit(EXIT_FAILURE);
    }
}

void iniciar_planificacion(t_scheduler_context* ctx) {
    log_info(ctx->logger, "Se debe iniciar el proceso con script: %s", ctx->path_script_inicial);
    
    t_proceso_planificable* pcb_inicial = malloc(sizeof(t_proceso_planificable));
    pcb_inicial->pid = 0; 
    pcb_inicial->cola_prioridad = 0;
    pcb_inicial->bloqueo_id = 0;
    
    enviar_peticion_crear_proceso(pcb_inicial->pid, ctx->path_script_inicial, ctx->socket_memoria);
    
    planificador_encolar_new(ctx->estratega, pcb_inicial);
    planificador_admitir_a_ready(ctx->estratega);
}

void iniciar_servidor_clientes(t_scheduler_context* ctx) {
    ctx->socket_escucha = iniciar_servidor(ctx->puerto_escucha, ctx->logger);
    if (ctx->socket_escucha == -1) {
        log_error(ctx->logger, "No se pudo iniciar el servidor en el puerto %s", ctx->puerto_escucha);
        limpiar_contexto(ctx);
        exit(EXIT_FAILURE);
    }

    log_info(ctx->logger, "Kernel Scheduler listo para recibir clientes en el puerto %s", ctx->puerto_escucha);

    while (1) {
        int cliente_fd = esperar_cliente(ctx->socket_escucha, ctx->logger);
        if (cliente_fd == -1) continue;
        
        t_args_hilo_scheduler* args = malloc(sizeof(t_args_hilo_scheduler));
        args->cliente_fd = cliente_fd;
        args->ctx = ctx;

        pthread_t hilo_cliente;
        if (pthread_create(&hilo_cliente, NULL, (void*)atender_cliente_scheduler, args) != 0) {
            log_error(ctx->logger, "Fallo al crear hilo cliente");
            free(args);
            close(cliente_fd);
        } else {
            pthread_detach(hilo_cliente);
        }
    }
}

void limpiar_contexto(t_scheduler_context* ctx) {
    if (ctx->socket_escucha != -1) close(ctx->socket_escucha);
    if (ctx->socket_memoria != -1) close(ctx->socket_memoria);

    if (ctx->estratega != NULL) destruir_estratega_planificacion(ctx->estratega);
    if (ctx->config != NULL) config_destroy(ctx->config);
    if (ctx->logger != NULL) log_destroy(ctx->logger);

    free(ctx);
}

    estratega->quantum_rr = config_get_int_value(config, "RR_QUANTUM");
    
    if (config_has_property(config, "SUSPENSION_TIMEOUT")) {
        estratega->suspension_timeout = config_get_int_value(config, "SUSPENSION_TIMEOUT");
    } else {
        estratega->suspension_timeout = 35000;
    }

    estratega->colas = crear_colas_ready_desde_config(config, estratega->quantum_rr, logger);
    estratega->proceso_exec = NULL;
    estratega->desalojo_entre_colas = config_get_bool_compatible(config, "QUEUE_PREEMPTION");
    estratega->logger = logger;
    estratega->socket_cpu = -1;
    pthread_mutex_init(&estratega->mutex_colas, NULL);

    if (estratega->algoritmo == ALGORITMO_PLANIFICACION_INVALIDO) {
        log_error(logger, "Algoritmo de planificacion invalido");
        destruir_estratega_planificacion(estratega);
        return NULL;
    }

    if (estratega->colas == NULL || list_is_empty(estratega->colas)) {
        log_error(logger, "No se configuro ninguna cola de planificacion");
        destruir_estratega_planificacion(estratega);
        return NULL;
    }

    log_info(logger, "Planificador inicializado: algoritmo=%s colas=%d desalojo=%s quantum=%d",
        planificador_nombre_algoritmo(estratega->algoritmo),
        list_size(estratega->colas),
        estratega->desalojo_entre_colas ? "TRUE" : "FALSE",
        estratega->quantum_rr);

    return estratega;
}

void destruir_mutex(void* element) {
    t_mutex* m = (t_mutex*)element;
    if (m) {
        free(m->nombre);
        queue_destroy(m->cola_bloqueados);
        pthread_mutex_destroy(&m->mutex_semaforo);
        free(m);
    }
}

void destruir_estratega_planificacion(t_estratega_planificacion* estratega) {
    if (estratega == NULL) return;

    queue_destroy(estratega->cola_new);

    if (estratega->colas != NULL) {
        list_destroy_and_destroy_elements(estratega->colas, destruir_cola_planificacion);
    }
    if (estratega->colas_io != NULL) {
        dictionary_destroy_and_destroy_elements(estratega->colas_io, (void*)queue_destroy);
    }
    if (estratega->mutex_creados != NULL) {
        dictionary_destroy_and_destroy_elements(estratega->mutex_creados, (void*)destruir_mutex); 
    }

    pthread_mutex_destroy(&estratega->mutex_colas);
    free(estratega);
}

bool planificador_encolar_new(t_estratega_planificacion* estratega, t_proceso_planificable* proceso) {
    if (estratega == NULL || proceso == NULL) {
        return false;
    }

    pthread_mutex_lock(&estratega->mutex_colas);
    proceso->estado = ESTADO_PROCESO_NEW;
    
    log_info(estratega->logger, "## (%d) Se crea el proceso - Estado: NEW", proceso->pid);

    queue_push(estratega->cola_new, proceso);
    pthread_mutex_unlock(&estratega->mutex_colas);
    return true;
}

t_proceso_planificable* planificador_admitir_a_ready(t_estratega_planificacion* estratega) {
    if (estratega == NULL) {
        return NULL;
    }

    pthread_mutex_lock(&estratega->mutex_colas);

    if (queue_is_empty(estratega->cola_new)) {
        pthread_mutex_unlock(&estratega->mutex_colas);
        return NULL;
    }

    t_proceso_planificable* proceso = queue_pop(estratega->cola_new);

    if (proceso->cola_prioridad < 0 || proceso->cola_prioridad >= list_size(estratega->colas)) {
        proceso->cola_prioridad = 0;
    }

    t_estado_proceso estado_anterior = proceso->estado;
    proceso->estado = ESTADO_PROCESO_READY;
    
    if (estado_anterior == ESTADO_PROCESO_SUSP_READY) {
        log_info(estratega->logger, "## Solicitando DESUSPENDER_PROCESO a Memoria para PID %d", proceso->pid);
        enviar_int(estratega->socket_memoria, DESUSPENDER_PROCESO);
        send(estratega->socket_memoria, &proceso->pid, sizeof(uint32_t), 0);
        
        recibir_mensaje(estratega->socket_memoria, estratega->logger);
    }

    log_info(estratega->logger, "## (%d) Pasa del estado %s al estado %s", 
             proceso->pid, 
             planificador_nombre_estado(estado_anterior), 
             planificador_nombre_estado(proceso->estado));

    encolar_en_cola_ready(estratega->colas, proceso);

    pthread_mutex_unlock(&estratega->mutex_colas);
    return proceso;
}

bool planificador_encolar(t_estratega_planificacion* estratega, t_proceso_planificable* proceso) {
    if (estratega == NULL || proceso == NULL) {
        return false;
    }

    pthread_mutex_lock(&estratega->mutex_colas);
    bool encolado = encolar_en_cola_ready(estratega->colas, proceso);

    if (!encolado) {
        pthread_mutex_unlock(&estratega->mutex_colas);
        return false;
    }

    pthread_mutex_unlock(&estratega->mutex_colas);
    return true;
}

t_proceso_planificable* planificador_obtener_siguiente(t_estratega_planificacion* estratega) {
    if (estratega == NULL) {
        return NULL;
    }

    pthread_mutex_lock(&estratega->mutex_colas);

    t_proceso_planificable* proceso = desencolar_siguiente_ready(estratega->colas);
    pthread_mutex_unlock(&estratega->mutex_colas);
    return proceso;
}

t_proceso_planificable* planificador_despachar_siguiente(t_estratega_planificacion* estratega) {
    t_proceso_planificable* proceso = planificador_obtener_siguiente(estratega);

    if (proceso == NULL) {
        return NULL;
    }

    pthread_mutex_lock(&estratega->mutex_colas);
    t_estado_proceso estado_anterior = proceso->estado;
    proceso->estado = ESTADO_PROCESO_EXEC;
    estratega->proceso_exec = proceso;
    
    log_info(estratega->logger, "## (%d) Pasa del estado %s al estado %s", 
             proceso->pid, 
             planificador_nombre_estado(estado_anterior), 
             planificador_nombre_estado(proceso->estado));

    pthread_mutex_unlock(&estratega->mutex_colas);

    return proceso;
}

void planificador_liberar_exec(t_estratega_planificacion* estratega, int pid) {
    if (estratega == NULL) {
        return;
    }

    pthread_mutex_lock(&estratega->mutex_colas);

    if (estratega->proceso_exec != NULL && estratega->proceso_exec->pid == pid) {
        estratega->proceso_exec = NULL;
    }

    pthread_mutex_unlock(&estratega->mutex_colas);
}

t_interfaz_io* buscar_interfaz_por_socket(t_estratega_planificacion* estratega, int socket_fd) {
    t_list* interfaces = dictionary_elements(estratega->colas_io);
    for (int i = 0; i < list_size(interfaces); i++) {
        t_interfaz_io* inter = list_get(interfaces, i);
        if (inter->socket_fd == socket_fd) {
            list_destroy(interfaces);
            return inter;
        }
    }
    list_destroy(interfaces);
    return NULL;
}

t_interfaz_io* buscar_interfaz_por_nombre(t_estratega_planificacion* estratega, char* nombre) {
    if (dictionary_has_key(estratega->colas_io, nombre)) {
        return dictionary_get(estratega->colas_io, nombre);
    }
    return NULL;
}

void planificador_registrar_io(t_estratega_planificacion* estratega, char* nombre_interfaz, int socket_fd) {
    pthread_mutex_lock(&estratega->mutex_colas);
    t_interfaz_io* nueva_io = malloc(sizeof(t_interfaz_io));
    nueva_io->nombre = string_duplicate(nombre_interfaz);
    nueva_io->socket_fd = socket_fd;
    nueva_io->solicitud_actual = NULL;
    nueva_io->cola_bloqueados = queue_create();
    pthread_mutex_init(&nueva_io->mutex_interfaz, NULL);
    dictionary_put(estratega->colas_io, nombre_interfaz, nueva_io);
    pthread_mutex_unlock(&estratega->mutex_colas);
    log_info(estratega->logger, "## Interfaz I/O %s registrada...", nombre_interfaz);
}

bool planificador_es_io_registrado(t_estratega_planificacion* estratega, int socket_fd) {
    if (estratega == NULL) return false;
    pthread_mutex_lock(&estratega->mutex_colas);
    t_interfaz_io* interfaz = buscar_interfaz_por_socket(estratega, socket_fd);
    pthread_mutex_unlock(&estratega->mutex_colas);
    return interfaz != NULL;
}

typedef struct {
    t_estratega_planificacion* estratega;
    t_proceso_planificable* pcb;
    int bloqueo_id;
} t_timer_args;

void* suspension_timer_thread(void* arg) {
    t_timer_args* a = (t_timer_args*)arg;
    usleep(a->estratega->suspension_timeout * 1000); // usleep takes microseconds
    
    pthread_mutex_lock(&a->estratega->mutex_colas);
    if (a->pcb->estado == ESTADO_PROCESO_BLOCK && a->pcb->bloqueo_id == a->bloqueo_id) {
        log_info(a->estratega->logger, "## Proceso %d suspendido por timeout", a->pcb->pid);
        a->pcb->estado = ESTADO_PROCESO_SUSP_BLOCK;
        
        enviar_int(a->estratega->socket_memoria, SUSPENDER_PROCESO);
        send(a->estratega->socket_memoria, &a->pcb->pid, sizeof(uint32_t), 0);
        
        recibir_mensaje(a->estratega->socket_memoria, a->estratega->logger);
    }
    pthread_mutex_unlock(&a->estratega->mutex_colas);
    free(a);
    return NULL;
}

void planificador_atender_peticion_io(t_estratega_planificacion* estratega, t_proceso_planificable* pcb, char* interfaz_nombre) {
    pthread_mutex_lock(&estratega->mutex_colas);
    t_interfaz_io* interfaz = buscar_interfaz_por_nombre(estratega, interfaz_nombre);
    
    if (interfaz == NULL) {
        log_error(estratega->logger, "La interfaz %s no existe", interfaz_nombre);
        pthread_mutex_unlock(&estratega->mutex_colas);
        return;
    }
    
    pthread_mutex_lock(&interfaz->mutex_interfaz);
    
    t_solicitud_io* solicitud = malloc(sizeof(t_solicitud_io));
    solicitud->proceso = pcb;
    solicitud->param1 = NULL;
    solicitud->param2 = NULL;
    
    pcb->estado = ESTADO_PROCESO_BLOCK;
    pcb->bloqueo_id++;
    log_info(estratega->logger, "## Proceso %d bloqueado esperando I/O en %s", pcb->pid, interfaz_nombre);

    // Iniciar hilo de timer para suspender proceso
    typedef struct {
        t_estratega_planificacion* estratega;
        t_proceso_planificable* pcb;
        int bloqueo_id;
    } t_timer_args;
    
    t_timer_args* args = malloc(sizeof(t_timer_args));
    args->estratega = estratega;
    args->pcb = pcb;
    args->bloqueo_id = pcb->bloqueo_id;
    
    pthread_t timer_thread;
    pthread_create(&timer_thread, NULL, (void*)suspension_timer_thread, args);
    pthread_detach(timer_thread);

    if (interfaz->solicitud_actual == NULL) {
        interfaz->solicitud_actual = solicitud;
    } else {
        queue_push(interfaz->cola_bloqueados, solicitud);
    }
    
    if (estratega->proceso_exec != NULL && estratega->proceso_exec->pid == pcb->pid) {
        estratega->proceso_exec = NULL;
    }
    
    pthread_mutex_unlock(&interfaz->mutex_interfaz);
    pthread_mutex_unlock(&estratega->mutex_colas);
}

void planificador_io_terminada(t_estratega_planificacion* estratega, int socket_fd) {
    pthread_mutex_lock(&estratega->mutex_colas);
    t_interfaz_io* interfaz = buscar_interfaz_por_socket(estratega, socket_fd);
    
    if (interfaz != NULL) {
        pthread_mutex_lock(&interfaz->mutex_interfaz);
        if (interfaz->solicitud_actual != NULL) {
            t_proceso_planificable* liberado = interfaz->solicitud_actual->proceso;
            log_info(estratega->logger, "## Proceso %d finalizó I/O en %s", liberado->pid, interfaz->nombre);
            free(interfaz->solicitud_actual);
            interfaz->solicitud_actual = NULL;
            
            t_estado_proceso estado_anterior = liberado->estado;
            if (estado_anterior == ESTADO_PROCESO_SUSP_BLOCK) {
                liberado->estado = ESTADO_PROCESO_SUSP_READY;
            } else {
                liberado->estado = ESTADO_PROCESO_READY;
            }
            
            log_info(estratega->logger, "## (%d) Pasa del estado %s al estado %s", 
                     liberado->pid, 
                     planificador_nombre_estado(estado_anterior), 
                     planificador_nombre_estado(liberado->estado));

            if (liberado->estado == ESTADO_PROCESO_SUSP_READY) {
                // If SUSP_READY, we should put it in the new processes queue or a SUSP_READY queue.
                // In TP, SUSP_READY goes to the new queue (cola_new) wait or directly to ready queue?
                // "Una vez que el proceso vuelva de este evento pasará al estado Ready / Suspended Ready."
                // "Si pasa a Suspended Ready debe ser agregado a la cola de News para que luego se pueda volver a llevar a Memoria Principal".
                // Wait, if it goes to New, we should push it to `estratega->cola_new` and call admitir_a_ready!
                queue_push(estratega->cola_new, liberado);
                log_info(estratega->logger, "Proceso %d movido a cola NEW desde SUSP_READY", liberado->pid);
                // planificador_admitir_a_ready(estratega);
            } else {
                encolar_en_cola_ready(estratega->colas, liberado);
            }
            
            if (!queue_is_empty(interfaz->cola_bloqueados)) {
                interfaz->solicitud_actual = queue_pop(interfaz->cola_bloqueados);
            }
        }
        pthread_mutex_unlock(&interfaz->mutex_interfaz);
    }
    pthread_mutex_unlock(&estratega->mutex_colas);
}

const char* planificador_nombre_algoritmo(t_algoritmo_planificacion algoritmo) {
    switch (algoritmo) {
        case ALGORITMO_PLANIFICACION_CMN:
            return "CMN";
        case ALGORITMO_PLANIFICACION_FIFO:
            return "FIFO";
        case ALGORITMO_PLANIFICACION_RR:
            return "RR";
        default:
            return "INVALIDO";
    }
}

void planificador_mutex_create(t_estratega_planificacion* estratega, t_proceso_planificable* pcb, char* recurso) {
    if (estratega == NULL || pcb == NULL || recurso == NULL) return;

    pthread_mutex_lock(&estratega->mutex_colas);
    if (!dictionary_has_key(estratega->mutex_creados, recurso)) {
        t_mutex* nuevo_mutex = malloc(sizeof(t_mutex));
        nuevo_mutex->nombre = string_duplicate(recurso);
        nuevo_mutex->tomado = 0;
        nuevo_mutex->pid_duenio = -1;
        nuevo_mutex->cola_bloqueados = queue_create();
        pthread_mutex_init(&nuevo_mutex->mutex_semaforo, NULL);

        dictionary_put(estratega->mutex_creados, recurso, nuevo_mutex);
        log_info(estratega->logger, "## Mutex Creado: %s", recurso);
    }
    pthread_mutex_unlock(&estratega->mutex_colas);
}

int planificador_mutex_lock(t_estratega_planificacion* estratega, t_proceso_planificable* pcb, char* recurso) {
    if (estratega == NULL || pcb == NULL || recurso == NULL) return 0;

    pthread_mutex_lock(&estratega->mutex_colas);
    t_mutex* mutex = dictionary_get(estratega->mutex_creados, recurso);
    if (mutex == NULL) {
        pthread_mutex_unlock(&estratega->mutex_colas);
        return 0; // Si no existe, no hacemos nada y sigue en EXEC
    }

    pthread_mutex_lock(&mutex->mutex_semaforo);
    if (mutex->tomado == 0) {
        // Libre: se lo damos al proceso actual
        mutex->tomado = 1;
        mutex->pid_duenio = pcb->pid;
        log_info(estratega->logger, "## Proceso %d toma el mutex %s", pcb->pid, recurso);
        pthread_mutex_unlock(&mutex->mutex_semaforo);
        pthread_mutex_unlock(&estratega->mutex_colas);
        return 0; // Retorna 0 (no se bloqueó, sigue en EXEC)
    } else {
        // Ocupado: a la cola de bloqueados del Mutex
        pcb->estado = ESTADO_PROCESO_BLOCK;
        queue_push(mutex->cola_bloqueados, pcb);
        log_info(estratega->logger, "## Proceso %d bloqueado esperando Mutex %s", pcb->pid, recurso);
        
        // Lo sacamos de EXEC
        if (estratega->proceso_exec != NULL && estratega->proceso_exec->pid == pcb->pid) {
            estratega->proceso_exec = NULL;
        }

        pthread_mutex_unlock(&mutex->mutex_semaforo);
        pthread_mutex_unlock(&estratega->mutex_colas);
        return 1; // Retorna 1 (se bloqueó)
    }
}

void planificador_mutex_unlock(t_estratega_planificacion* estratega, t_proceso_planificable* pcb, char* recurso) {
    if (estratega == NULL || pcb == NULL || recurso == NULL) return;

    pthread_mutex_lock(&estratega->mutex_colas);
    t_mutex* mutex = dictionary_get(estratega->mutex_creados, recurso);
    if (mutex == NULL) {
        pthread_mutex_unlock(&estratega->mutex_colas);
        return;
    }

    pthread_mutex_lock(&mutex->mutex_semaforo);
    if (mutex->tomado == 1 && mutex->pid_duenio == pcb->pid) {
        log_info(estratega->logger, "## Proceso %d libera Mutex %s", pcb->pid, recurso);
        
        // Vemos si hay alguien esperando
        if (!queue_is_empty(mutex->cola_bloqueados)) {
            t_proceso_planificable* liberado = queue_pop(mutex->cola_bloqueados);
            // Le damos la tenencia del Mutex al desencolado
            mutex->pid_duenio = liberado->pid;
            
            log_info(estratega->logger, "## Proceso %d desencolado y asume Mutex %s", liberado->pid, recurso);
            
            // Lo mandamos a READY
            t_estado_proceso estado_anterior = liberado->estado;
            liberado->estado = ESTADO_PROCESO_READY;
            
            log_info(estratega->logger, "## (%d) Pasa del estado %s al estado %s", 
                     liberado->pid, 
                     planificador_nombre_estado(estado_anterior), 
                     planificador_nombre_estado(liberado->estado));

            encolar_en_cola_ready(estratega->colas, liberado);
        } else {
            // Nadie esperaba, el mutex queda libre
            mutex->tomado = 0;
            mutex->pid_duenio = -1;
        }
    }
    pthread_mutex_unlock(&mutex->mutex_semaforo);
    pthread_mutex_unlock(&estratega->mutex_colas);
}

#include "rr.h"

static void dormir_milisegundos(int milisegundos) {
    struct timespec tiempo;
    tiempo.tv_sec = milisegundos / 1000;
    tiempo.tv_nsec = (milisegundos % 1000) * 1000000L;
    nanosleep(&tiempo, NULL);
}

static void* controlar_quantum(void* args_void) {
    t_args_quantum* args = args_void;

    dormir_milisegundos(args->quantum_ms);

    pthread_mutex_lock(&args->estratega->mutex_colas);
    bool sigue_en_exec = args->estratega->proceso_exec != NULL
        && args->estratega->proceso_exec->pid == args->pid;
    pthread_mutex_unlock(&args->estratega->mutex_colas);

    if (sigue_en_exec) {
        t_paquete* paquete = crear_paquete();
        paquete->codigo_operacion = INTERRUPCION_QUANTUM;

        enviar_paquete(paquete, args->socket_cpu);

        eliminar_paquete(paquete);

        log_info(args->estratega->logger, "## (%d) - Desalojado por fin de quantum", args->pid);
    }
    free(args);
    return NULL;
}

bool rr_iniciar_quantum(t_estratega_planificacion* estratega, t_proceso_planificable* proceso, int socket_cpu) {
    int quantum = planificador_quantum_para(estratega, proceso != NULL ? proceso->cola_prioridad : -1);

    if (estratega == NULL || proceso == NULL || socket_cpu < 0 || quantum <= 0) {
        return false;
    }

    t_args_quantum* args = malloc(sizeof(t_args_quantum));
    args->estratega = estratega;
    args->pid = proceso->pid;
    args->socket_cpu = socket_cpu;
    args->quantum_ms = quantum;

    pthread_t hilo_quantum;
    if (pthread_create(&hilo_quantum, NULL, controlar_quantum, args) != 0) {
        free(args);
        return false;
    }

    pthread_detach(hilo_quantum);
    return true;
}

void* atender_cliente_cpu(void* arg) {
    t_args_hilo_cpu* args = (t_args_hilo_cpu*)arg;
    int socket_cpu = args->socket_cliente;
    t_memory_stick_context* ctx = args->ctx;
    free(args);

    while (1) {
        int cod_op = recibir_operacion(socket_cpu);

        switch (cod_op) {
            case MENSAJE:
                procesar_mensaje_handshake(socket_cpu, ctx->logger);
                break;
            case LEER_MEMORIA:
                procesar_lectura(socket_cpu, ctx);
                break;
            case ESCRIBIR_MEMORIA:
                procesar_escritura(socket_cpu, ctx);
                break;
            case -1:
                log_info(ctx->logger, "Cliente (CPU) desconectado");
                close(socket_cpu);
                return NULL;
            default:
                log_warning(ctx->logger, "Operación desconocida de la CPU: %d", cod_op);
                close(socket_cpu);
                return NULL;
        }
    }
}

void procesar_mensaje_handshake(int socket_cpu, t_log* logger) {
    char *id_cpu = recibir_handshake(socket_cpu);
    log_info(logger, "## CPU %s Conectada", id_cpu);
    free(id_cpu);
}

void procesar_lectura(int socket_cpu, t_memory_stick_context* ctx) {
    int size;
    void* buffer = recibir_buffer(&size, socket_cpu);
    t_peticion_lectura* peticion = deserializar_peticion_lectura(buffer);
    free(buffer);

    void* datos_leidos = malloc(peticion->tamanio);
    
    usleep(ctx->memory_delay * 1000);
    
    // Lectura sincronizada con mutex
    pthread_mutex_lock(&ctx->mutex_memoria);
    if (peticion->direccion_fisica + peticion->tamanio <= ctx->tamanio_total) {
        memcpy(datos_leidos, ctx->espacio_memoria + peticion->direccion_fisica, peticion->tamanio);
        log_info(ctx->logger, "## Lectura de %u bytes", peticion->tamanio);
    } else {
        log_error(ctx->logger, "Error: Intento de lectura fuera de los límites de la memoria");
        memset(datos_leidos, 0, peticion->tamanio);
    }
    pthread_mutex_unlock(&ctx->mutex_memoria);

    // Devolver resultado
    t_paquete* paquete = crear_paquete();
    paquete->codigo_operacion = LECTURA;
    agregar_a_paquete(paquete, datos_leidos, peticion->tamanio);
    enviar_paquete(paquete, socket_cpu);
    
    eliminar_paquete(paquete);
    free(datos_leidos);
    destruir_peticion_lectura(peticion);
}

void procesar_escritura(int socket_cpu, t_memory_stick_context* ctx) {
    int size;
    void* buffer = recibir_buffer(&size, socket_cpu);
    t_peticion_escritura* peticion = deserializar_peticion_escritura(buffer);
    free(buffer);
    
    usleep(ctx->memory_delay * 1000);

    // Escritura sincronizada con mutex
    pthread_mutex_lock(&ctx->mutex_memoria);
    if (peticion->direccion_fisica + peticion->tamanio <= ctx->tamanio_total) {
        memcpy(ctx->espacio_memoria + peticion->direccion_fisica, peticion->contenido, peticion->tamanio);
        log_info(ctx->logger, "## Escritura de %u bytes", peticion->tamanio);
    } else {
        log_error(ctx->logger, "Error: Intento de escritura fuera de los límites de la memoria");
    }
    pthread_mutex_unlock(&ctx->mutex_memoria);

    // Responder OK
    enviar_mensaje(MSG_OK, socket_cpu);
    
    destruir_peticion_escritura(peticion);
}

t_memory_stick_context* iniciar_contexto(int argc, char* argv[]) {
    t_memory_stick_context* ctx = malloc(sizeof(t_memory_stick_context));
    
    if (argc != 3) {
        printf("Uso: ./bin/memory_stick [Config] [Tamaño]\n");
        free(ctx);
        exit(EXIT_FAILURE);
    }
    
    char* path_config = argv[1];
    uint32_t tamanio_stick = (uint32_t)atoi(argv[2]);

    if (tamanio_stick <= 0) {
        printf("Error: El tamaño del Memory Stick debe ser mayor a 0\n");
        free(ctx);
        exit(EXIT_FAILURE);
    }

    ctx->config = config_create(path_config);
    if (ctx->config == NULL) {
        printf("No se pudo cargar la configuración desde %s\n", path_config);
        free(ctx);
        exit(EXIT_FAILURE);
    }
    
    char* log_level_str = config_get_string_value(ctx->config, "LOG_LEVEL");
    t_log_level level = LOG_LEVEL_INFO;
    if (log_level_str != NULL) {
        if (string_equals_ignore_case(log_level_str, "TRACE")) level = LOG_LEVEL_TRACE;
        else if (string_equals_ignore_case(log_level_str, "DEBUG")) level = LOG_LEVEL_DEBUG;
        else if (string_equals_ignore_case(log_level_str, "INFO")) level = LOG_LEVEL_INFO;
        else if (string_equals_ignore_case(log_level_str, "WARNING")) level = LOG_LEVEL_WARNING;
        else if (string_equals_ignore_case(log_level_str, "ERROR")) level = LOG_LEVEL_ERROR;
    }
    
    ctx->logger = log_create("memory_stick.log", "MEMORY_STICK", true, level);

    ctx->tamanio_total = tamanio_stick;
    ctx->espacio_memoria = malloc(tamanio_stick);
    if (ctx->espacio_memoria == NULL) {
        log_error(ctx->logger, "Error: No se pudo asignar %u bytes para el Memory Stick", tamanio_stick);
        config_destroy(ctx->config);
        log_destroy(ctx->logger);
        free(ctx);
        exit(EXIT_FAILURE);
    }
    memset(ctx->espacio_memoria, 0, tamanio_stick); // Inicializar en 0

    pthread_mutex_init(&ctx->mutex_memoria, NULL);

    ctx->puerto_escucha = config_get_string_value(ctx->config, "PUERTO_STICK");
    ctx->ip_memory = config_get_string_value(ctx->config, "IP_MEMORY");
    ctx->puerto_memory = config_get_string_value(ctx->config, "PUERTO_MEMORY");
    ctx->memory_delay = config_get_int_value(ctx->config, "MEMORY_DELAY");
    ctx->socket_kernel_memory = -1;
    ctx->socket_escucha = -1;

    return ctx;
}

void conectar_kernel_memory(t_memory_stick_context* ctx) {
    ctx->socket_kernel_memory = crear_conexion(ctx->ip_memory, ctx->puerto_memory);
    if (ctx->socket_kernel_memory == -1) {
        log_error(ctx->logger, "No se pudo conectar a Kernel Memory en %s:%s", ctx->ip_memory, ctx->puerto_memory);
        limpiar_contexto(ctx);
        exit(EXIT_FAILURE);
    }
    log_info(ctx->logger, "## Conectado a Kernel Memory");

    // Mandar Handshake indicando el tamaño, IP y puerto de escucha
    // Asumimos que IP_MEMORY u otra prop nos sirve. Pero el memory_stick no tiene su propia IP en config, usa 127.0.0.1 por defecto localmente?
    // En las configs suele haber "IP_MEMORY", usamos esa o 127.0.0.1
    char *msj_handshake = string_from_format("%s|%u|%s|%s", HANDSHAKE_STICK, ctx->tamanio_total, "127.0.0.1", ctx->puerto_escucha);
    enviar_mensaje(msj_handshake, ctx->socket_kernel_memory);
    free(msj_handshake);
}

void iniciar_servidor_cpus(t_memory_stick_context* ctx) {
    ctx->socket_escucha = iniciar_servidor(ctx->puerto_escucha, ctx->logger);
    if (ctx->socket_escucha == -1) {
        log_error(ctx->logger, "Fallo al iniciar el servidor del Memory Stick en el puerto %s", ctx->puerto_escucha);
        limpiar_contexto(ctx);
        exit(EXIT_FAILURE);
    }
    log_info(ctx->logger, "Memory Stick listo para recibir CPUs en el puerto %s", ctx->puerto_escucha);

    while (1) {
        int socket_cpu = esperar_cliente(ctx->socket_escucha, ctx->logger);
        if (socket_cpu == -1) {
            log_warning(ctx->logger, "Error al aceptar una nueva conexión de CPU");
            continue;
        }

        t_args_hilo_cpu* args = malloc(sizeof(t_args_hilo_cpu));
        args->socket_cliente = socket_cpu;
        args->ctx = ctx;

        pthread_t hilo_cpu;
        if (pthread_create(&hilo_cpu, NULL, atender_cliente_cpu, args) != 0) {
            log_error(ctx->logger, "Fallo al crear el hilo para la CPU");
            free(args);
            close(socket_cpu);
        } else {
            pthread_detach(hilo_cpu); // Permite al hilo liberar sus propios recursos al terminar
        }
    }
}

void limpiar_contexto(t_memory_stick_context* ctx) {
    if (ctx->socket_escucha != -1) close(ctx->socket_escucha);
    if (ctx->socket_kernel_memory != -1) close(ctx->socket_kernel_memory);
    
    if (ctx->espacio_memoria != NULL) {
        free(ctx->espacio_memoria);
    }
    pthread_mutex_destroy(&ctx->mutex_memoria);
    
    if (ctx->config != NULL) config_destroy(ctx->config);
    if (ctx->logger != NULL) log_destroy(ctx->logger);
    
    free(ctx);
}

void crear_archivo_swap(t_swap_context* ctx) {
    ctx->swap_file = fopen(ctx->swap_file_path, "wb+");
    
    if (ctx->swap_file == NULL) {
        log_error(ctx->logger, "No se pudo crear el archivo de SWAP");
        limpiar_contexto(ctx);
        exit(EXIT_FAILURE);
    }

    // Inicializamos el archivo truncándolo al tamaño especificado
    fseek(ctx->swap_file, ctx->swap_file_size - 1, SEEK_SET);
    fputc('\0', ctx->swap_file);
}

void conectar_kernel_memory(t_swap_context* ctx) {
    ctx->socket_kernel_memory = crear_conexion(ctx->ip_memory, ctx->puerto_memory);

    if (ctx->socket_kernel_memory == -1) {
        log_error(ctx->logger, "No se pudo conectar a Kernel Memory");
        limpiar_contexto(ctx);
        exit(EXIT_FAILURE);
    }
    
    log_info(ctx->logger, "## Conectado a Kernel Memory");

    // Handshake
    char* msj_handshake = string_from_format("SWAP|%d|%d", ctx->swap_file_size, ctx->block_size);
    enviar_mensaje(msj_handshake, ctx->socket_kernel_memory);
    free(msj_handshake);
    
    log_info(ctx->logger, "Mensaje de handshake enviado a Kernel Memory correctamente");
}

void procesar_peticiones_memoria(t_swap_context* ctx) {
    while(1) {
        log_info(ctx->logger, "Esperando peticiones de Kernel Memory...");
        int cod_op = recibir_operacion(ctx->socket_kernel_memory);
        
        switch(cod_op) {
             case LECTURA:
                  procesar_peticion_lectura(ctx);
                  break;
             case ESCRITURA:
                  procesar_peticion_escritura(ctx);
                  break;
             case -1:
                  log_error(ctx->logger, "Desconexión de Kernel Memory");
                  return; // Sale del bucle para ir a limpiar_contexto()
             default:
                  log_warning(ctx->logger, "Operacion desconocida: %d", cod_op);
                  break;
         }
    }
}

void limpiar_contexto(t_swap_context* ctx) {
    if (ctx->swap_file != NULL) {
        fclose(ctx->swap_file); 
    }
    if (ctx->socket_kernel_memory != -1) {
        close(ctx->socket_kernel_memory);
    }
    if (ctx->config != NULL) {
        config_destroy(ctx->config);
    }
    if (ctx->logger != NULL) {
        log_destroy(ctx->logger);
    }
    free(ctx);
}

#include "peticiones_swap.h"

void procesar_peticion_lectura(t_swap_context* ctx) {
    int numero_bloque = recibir_operacion(ctx->socket_kernel_memory);
    
    log_info(ctx->logger, "## Lectura del bloque: %d", numero_bloque);
    
    void* buffer = malloc(ctx->block_size);
    
    fseek(ctx->swap_file, numero_bloque * ctx->block_size, SEEK_SET);
    fread(buffer, 1, ctx->block_size, ctx->swap_file);
    
    t_paquete* paq = crear_paquete();
    paq->codigo_operacion = LECTURA;
    agregar_a_paquete(paq, buffer, ctx->block_size);
    enviar_paquete(paq, ctx->socket_kernel_memory);
    eliminar_paquete(paq);
    
    free(buffer);
}

void procesar_peticion_escritura(t_swap_context* ctx) {
    int numero_bloque = recibir_operacion(ctx->socket_kernel_memory);
    
    int size;
    void* buffer = recibir_buffer(&size, ctx->socket_kernel_memory);
    
    log_info(ctx->logger, "## Escritura del bloque: %d", numero_bloque);
    
    fseek(ctx->swap_file, numero_bloque * ctx->block_size, SEEK_SET);
    fwrite(buffer, 1, ctx->block_size, ctx->swap_file);
    
    // Es importante hacer flush o fclose no, pero un flush asegura que se guarde en disco
    fflush(ctx->swap_file);
    
    enviar_mensaje("OK", ctx->socket_kernel_memory);
    free(buffer);
}
