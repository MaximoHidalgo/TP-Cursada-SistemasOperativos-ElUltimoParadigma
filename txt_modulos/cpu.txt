#include "ciclo_instruccion.h"

void ejecutar_ciclo_de_instruccion(t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    int desalojar = 0;

    while (!desalojar) {

        /* 1. FETCH */ 
        char* instruccion_texto = fetch(contexto, ctx);
        
        if (instruccion_texto == NULL) {
            log_error(ctx->logger, "Fetch falló (probablemente Memoria se desconectó). Desalojando PID %d.", contexto->pid);
            break;
        }

        /* 2. DECODE */
        t_instruccion* instruccion_decodificada = decode(instruccion_texto);
        
        /* 3. EXECUTE */
        desalojar = execute(instruccion_decodificada, contexto, ctx);
        
        // Incrementamos el PC solo si no hubo desalojo y la instrucción NO fue un JNZ
        if (!desalojar && !string_equals_ignore_case(instruccion_decodificada->identificador, "JNZ")) {
            contexto->registros.PC++;
        }

        /* 4. CHECK INTERRUPT */
        if (!desalojar) {
            int hubo_interrupcion = chequear_interrupciones(ctx);
            if (hubo_interrupcion) {
                log_info(ctx->logger, "## Interrupción por Fin de Quantum recibida para PID %d", contexto->pid);
                desalojar = 1; // Obligamos a romper el ciclo de instrucción
                
                enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
                notificar_desalojo_a_scheduler(contexto->pid, FIN_DE_QUANTUM, ctx);
            }
        }

        // Limpiamos memoria (y también liberar la instrucción y sus parámetros)
        free(instruccion_texto);
        if (instruccion_decodificada->identificador) free(instruccion_decodificada->identificador);
        if (instruccion_decodificada->parametro1) free(instruccion_decodificada->parametro1);
        if (instruccion_decodificada->parametro2) free(instruccion_decodificada->parametro2);
        if (instruccion_decodificada->parametro3) free(instruccion_decodificada->parametro3);
        free(instruccion_decodificada);
    }
}

void notificar_desalojo_a_scheduler(uint32_t pid, int motivo_desalojo, t_cpu_context* ctx) {
    // 1. Creamos un paquete. Usamos el 'motivo_desalojo' directamente como el op_code
    t_paquete* paquete = crear_paquete();
    paquete->codigo_operacion = motivo_desalojo; 
    
    // 2. Agregamos únicamente el PID al paquete
    agregar_a_paquete(paquete, &pid, sizeof(uint32_t));
    
    // 3. Enviamos el paquete por el socket conectado al Scheduler
    enviar_paquete(paquete, ctx->conexiones->scheduler);
    
    // 4. Liberamos el paquete para evitar memory leaks
    eliminar_paquete(paquete);

    log_info(ctx->logger, "PID %d devuelto al Scheduler. Motivo (op_code): %d", pid, motivo_desalojo);
}

void notificar_syscall_sleep_a_scheduler(uint32_t pid, uint32_t tiempo_sleep, t_cpu_context* ctx) {
    t_paquete* paquete = crear_paquete();
    paquete->codigo_operacion = SYSCALL_SLEEP; 
    
    agregar_a_paquete(paquete, &pid, sizeof(uint32_t));
    agregar_a_paquete(paquete, &tiempo_sleep, sizeof(uint32_t)); // ¡Dato extra!
    
    enviar_paquete(paquete, ctx->conexiones->scheduler);
    eliminar_paquete(paquete);
}

int mmu_traducir(t_cpu_context* ctx, t_contexto_ejecucion* contexto, int dir_logica, int tam_bytes, int tam_max_segmento, uint32_t* dir_fisica) {
    int num_segmento = floor((double)dir_logica / tam_max_segmento);
    int desplazamiento = dir_logica % tam_max_segmento;

    t_segmento* seg = NULL;
    if (contexto->tabla_segmentos != NULL) {
        for (int i = 0; i < list_size(contexto->tabla_segmentos); i++) {
            t_segmento* s = list_get(contexto->tabla_segmentos, i);
            if (s->id == num_segmento) {
                seg = s;
                break;
            }
        }
    }

    if (!seg) return -1; // SEG FAULT

    if (desplazamiento + tam_bytes > seg->tamanio) {
        return -1; // SEG FAULT
    }

    *dir_fisica = seg->base + desplazamiento;
    return 0;
}

t_contexto_ejecucion* pedir_contexto_a_memoria(int pid_recibido, t_cpu_context* ctx) {
    t_paquete* paquete_solicitud = crear_paquete();
    paquete_solicitud->codigo_operacion = SOLICITUD_CONTEXTO; 
    
    //Agregamos el PID al paquete y lo enviamos
    agregar_a_paquete(paquete_solicitud, &pid_recibido, sizeof(int));
    enviar_paquete(paquete_solicitud, ctx->conexiones->memoria);
    eliminar_paquete(paquete_solicitud);

    log_info(ctx->logger, "Solicitando contexto del PID %d a Kernel Memory...", pid_recibido);

    //Quedarnos esperando la respuesta de la Memoria
    int operacion = recibir_operacion(ctx->conexiones->memoria);

    if (operacion == ENVIO_CONTEXTO) {
        int size;
        void* buffer = recibir_buffer(&size, ctx->conexiones->memoria);

        //Deserializar el buffer recibido para armar el struct localmente
        t_contexto_ejecucion* contexto = malloc(sizeof(t_contexto_ejecucion));
        deserializar_contexto(buffer, size, contexto);
        free(buffer);

        int cant = contexto->tabla_segmentos ? list_size(contexto->tabla_segmentos) : 0;
        log_info(ctx->logger, "Contexto de ejecución del PID %d recibido exitosamente con %d segmentos.", contexto->pid, cant);
        for(int i=0; i<cant; i++) {
            t_segmento* s = list_get(contexto->tabla_segmentos, i);
            log_info(ctx->logger, "  -> Seg ID: %d, Base: %d, Tamanio: %d", s->id, s->base, s->tamanio);
        }

        return contexto;
    } else {
        log_error(ctx->logger, "Error: Se esperaba el contexto de ejecución, pero llegó la operación: %d", operacion);
        return NULL;
    }
}

char* fetch(t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    //Creamos un paquete y le cargamos el OP Code FETCH_INSTRUCCION
    t_paquete* paquete_solicitud = crear_paquete();
    paquete_solicitud->codigo_operacion = FETCH_INSTRUCCION; 
    
    enviar_peticion_fetch_instruccion(contexto->pid, contexto->registros.PC, ctx->conexiones->memoria);

    int op_code = recibir_operacion(ctx->conexiones->memoria);
    
    if (op_code == ENVIO_INSTRUCCION) {
        t_list* lista = recibir_paquete(ctx->conexiones->memoria);
        char* instruccion_texto = string_duplicate(list_get(lista, 0));
        list_destroy_and_destroy_elements(lista, free);
        return instruccion_texto;
    } else {
        log_error(ctx->logger, "Error: Se esperaba una instrucción, pero llegó la operación: %d", op_code);
        return NULL;
    }
}

t_instruccion* decode(char* instruccion_texto) {
    t_instruccion* instruccion = malloc(sizeof(t_instruccion));
    instruccion->parametro1 = NULL;
    instruccion->parametro2 = NULL;
    instruccion->parametro3 = NULL;

    char** instruccion_separada = string_split(instruccion_texto, " ");

    // El primer string siempre es el comando a ejecutar
    instruccion->identificador = string_duplicate(instruccion_separada[0]);

    if (instruccion_separada[1] != NULL) {
    instruccion->parametro1 = string_duplicate(instruccion_separada[1]);
        if (instruccion_separada[2] != NULL) {
        instruccion->parametro2 = string_duplicate(instruccion_separada[2]);
            if (instruccion_separada[3] != NULL) {
                instruccion->parametro3 = string_duplicate(instruccion_separada[3]);
            }
        }
    }

    // Liberamos el array de strings que generó las Commons para evitar memory leaks
    string_array_destroy(instruccion_separada);

    return instruccion;
}

int execute(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    log_info(ctx->logger, "## PID: %d - Ejecutando: %s - %s %s %s", 
             contexto->pid, 
             instruccion->identificador, 
             instruccion->parametro1 ? instruccion->parametro1 : "", 
             instruccion->parametro2 ? instruccion->parametro2 : "", 
             instruccion->parametro3 ? instruccion->parametro3 : "");

    t_instruction *inst = instruction_get_by_name(instruccion->identificador);
    
    if (inst != NULL) {
        return inst->instr(instruccion, contexto, ctx);
    } else {
        log_error(ctx->logger, "Instrucción desconocida: %s", instruccion->identificador);
        return 1; // Desalojar
    }
}

int chequear_interrupciones(t_cpu_context* ctx) {
    int op_code_espiado;
    
    // Miramos si hay datos en el socket de forma no bloqueante
    ssize_t bytes_recibidos = recv(ctx->conexiones->scheduler, &op_code_espiado, sizeof(int), MSG_PEEK | MSG_DONTWAIT);

    if (bytes_recibidos > 0) {
        int operacion = recibir_operacion(ctx->conexiones->scheduler);
        
        if (operacion == INTERRUPCION_QUANTUM) { 
            // Consumimos el payload del paquete para no dejar basura en el buffer del socket
            int size;
            void* buffer = recibir_buffer(&size, ctx->conexiones->scheduler);
            free(buffer);

            log_info(ctx->logger, "## Interrupción recibida");
            return 1;
        } else {
            log_warning(ctx->logger, "Se recibió un código inesperado durante el ciclo: %d", operacion);
        }
    } else if (bytes_recibidos == -1 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
        // No hay mensajes pendientes y la CPU sigue ejecutando tranquila.
        return 0;
    } else if (bytes_recibidos == 0) {
        log_error(ctx->logger, "El Kernel Scheduler se ha desconectado inesperadamente.");
        return 1; // Frenamos la CPU porque se cayó el servidor
    }
    return 0;
}

void* leer_memoria_distribuida_cpu(t_cpu_context* ctx, uint32_t dir_fisica, uint32_t tamanio) {
    void* buffer = malloc(tamanio);
    uint32_t leido = 0;
    
    while (leido < tamanio) {
        uint32_t dir_actual = dir_fisica + leido;
        t_memory_stick_info* stick_target = NULL;
        
        for (int i = 0; i < list_size(ctx->memory_sticks); i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            if (dir_actual >= stick->base_address && dir_actual < stick->base_address + stick->tamanio) {
                stick_target = stick;
                break;
            }
        }
        
        if (stick_target == NULL) {
            log_error(ctx->logger, "Error: Direccion fisica fuera de los limites de la memoria distribuida (CPU)");
            break;
        }
        
        uint32_t offset_en_stick = dir_actual - stick_target->base_address;
        uint32_t bytes_disponibles = stick_target->tamanio - offset_en_stick;
        uint32_t bytes_a_leer = (tamanio - leido < bytes_disponibles) ? (tamanio - leido) : bytes_disponibles;
        
        enviar_peticion_lectura(offset_en_stick, bytes_a_leer, stick_target->socket_fd);
        
        int r_op = recibir_operacion(stick_target->socket_fd);
        if (r_op == LECTURA) {
            int size_buf;
            void* res = recibir_buffer(&size_buf, stick_target->socket_fd);
            memcpy(buffer + leido, res, bytes_a_leer);
            free(res);
        } else {
            log_error(ctx->logger, "Error al leer del memory stick en CPU");
            break;
        }
        
        leido += bytes_a_leer;
    }
    
    return buffer;
}

void escribir_memoria_distribuida_cpu(t_cpu_context* ctx, uint32_t dir_fisica, uint32_t tamanio, void* buffer) {
    uint32_t escrito = 0;
    
    while (escrito < tamanio) {
        uint32_t dir_actual = dir_fisica + escrito;
        t_memory_stick_info* stick_target = NULL;
        
        for (int i = 0; i < list_size(ctx->memory_sticks); i++) {
            t_memory_stick_info* stick = list_get(ctx->memory_sticks, i);
            if (dir_actual >= stick->base_address && dir_actual < stick->base_address + stick->tamanio) {
                stick_target = stick;
                break;
            }
        }
        
        if (stick_target == NULL) {
            log_error(ctx->logger, "Error: Direccion fisica fuera de los limites de la memoria distribuida (CPU)");
            break;
        }
        
        uint32_t offset_en_stick = dir_actual - stick_target->base_address;
        uint32_t bytes_disponibles = stick_target->tamanio - offset_en_stick;
        uint32_t bytes_a_escribir = (tamanio - escrito < bytes_disponibles) ? (tamanio - escrito) : bytes_disponibles;
        
        enviar_peticion_escritura(offset_en_stick, bytes_a_escribir, buffer + escrito, stick_target->socket_fd);
        
        int r_op = recibir_operacion(stick_target->socket_fd);
        if (r_op == MENSAJE) {
            int size_buf;
            void* res = recibir_buffer(&size_buf, stick_target->socket_fd);
            free(res); // Esperamos MSG_OK
        } else {
            log_error(ctx->logger, "Error al escribir en memory stick en CPU");
            break;
        }
        
        escrito += bytes_a_escribir;
    }
}

    for (uint8_t i = 0; INSTRUCTION_SET[i].name != NULL; i++) {
        if (string_equals_ignore_case(INSTRUCTION_SET[i].name, name)) {
            return &INSTRUCTION_SET[i];

    return NULL;

int instr_sum(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t destino_val = leer_registro(contexto, instruccion->parametro1);
    uint32_t origen_val = leer_registro(contexto, instruccion->parametro2);
    asignar_valor_a_registro(contexto, instruccion->parametro1, destino_val + origen_val);
    return 0;
}

    uint32_t destino_val = leer_registro(contexto, instruccion->parametro1);
    uint32_t origen_val = leer_registro(contexto, instruccion->parametro2);
    asignar_valor_a_registro(contexto, instruccion->parametro1, destino_val - origen_val);
    return 0;
}

int instr_jnz(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t registro_val = leer_registro(contexto, instruccion->parametro1);

    if (registro_val != 0) {
        contexto->registros.PC = atoi(instruccion->parametro2);
    } else {
        contexto->registros.PC++;
    }
    return 0;
}

int instr_mov_in(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    char* reg_datos = instruccion->parametro1;
    int tam = obtener_tamanio_registro(reg_datos);
    uint32_t dir_logica = contexto->registros.SI;
    uint32_t dir_fisica = 0;

    if (mmu_traducir(ctx, contexto, dir_logica, tam, ctx->tam_max_segmento, &dir_fisica) == -1) {
        log_error(ctx->logger, "Segmentation Fault accediendo a dir logica %d", dir_logica);
        enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = SYSCALL_SEG_FAULT;
        agregar_a_paquete(paq, &(contexto->pid), sizeof(uint32_t));
        enviar_paquete(paq, ctx->conexiones->scheduler);
        eliminar_paquete(paq);
        return 1; // Desalojar
    } else {
        void* buf = leer_memoria_distribuida_cpu(ctx, dir_fisica, tam);
        if (buf != NULL) {
            uint32_t valor = 0;
            memcpy(&valor, buf, tam);
            free(buf);
            asignar_valor_a_registro(contexto, reg_datos, valor);
            log_info(ctx->logger, "PID: %d - Acción: LEER - Dirección Física: %d - Valor: %d", contexto->pid, dir_fisica, valor);
        }
        return 0;
    }
}

int instr_mov_out(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    char* reg_datos = instruccion->parametro1;
    int tam = obtener_tamanio_registro(reg_datos);
    uint32_t valor = leer_registro(contexto, reg_datos);
    uint32_t dir_logica = contexto->registros.DI;
    uint32_t dir_fisica = 0;

    if (mmu_traducir(ctx, contexto, dir_logica, tam, ctx->tam_max_segmento, &dir_fisica) == -1) {
        log_error(ctx->logger, "Segmentation Fault escribiendo a dir logica %d", dir_logica);
        enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = SYSCALL_SEG_FAULT;
        agregar_a_paquete(paq, &(contexto->pid), sizeof(uint32_t));
        enviar_paquete(paq, ctx->conexiones->scheduler);
        eliminar_paquete(paq);
        return 1; // Desalojar
    } else {
        escribir_memoria_distribuida_cpu(ctx, dir_fisica, tam, &valor);
        log_info(ctx->logger, "PID: %d - Acción: ESCRIBIR - Dirección Física: %d - Valor: %d", contexto->pid, dir_fisica, valor);
        return 0;
    }
}

int instr_copy_mem(t_instruccion* instruccion, t_contexto_ejecucion* contexto, t_cpu_context* ctx) {
    uint32_t tam = atoi(instruccion->parametro1);
    uint32_t dir_logica_si = contexto->registros.SI;
    uint32_t dir_logica_di = contexto->registros.DI;
    uint32_t dir_fisica_si = 0;
    uint32_t dir_fisica_di = 0;

    if (mmu_traducir(ctx, contexto, dir_logica_si, tam, ctx->tam_max_segmento, &dir_fisica_si) == -1 || 
        mmu_traducir(ctx, contexto, dir_logica_di, tam, ctx->tam_max_segmento, &dir_fisica_di) == -1) {
        log_error(ctx->logger, "Segmentation Fault en COPY_MEM");
        enviar_contexto_a_memoria(contexto, ctx->conexiones->memoria);
        t_paquete* paq = crear_paquete();
        paq->codigo_operacion = SYSCALL_SEG_FAULT;
        agregar_a_paquete(paq, &(contexto->pid), sizeof(uint32_t));
        enviar_paquete(paq, ctx->conexiones->scheduler);
        eliminar_paquete(paq);
        return 1;
    } else {
        void* buf = leer_memoria_distribuida_cpu(ctx, dir_fisica_si, tam);
        if (buf != NULL) {
            escribir_memoria_distribuida_cpu(ctx, dir_fisica_di, tam, buf);
            log_info(ctx->logger, "PID: %d - Acción: COPY_MEM - Origen Fisico: %d - Destino Fisico: %d", contexto->pid, dir_fisica_si, dir_fisica_di);
            free(buf);
        }
        return 0;
    }
}

    usleep(1000 * 100); // 100ms provisorio, podés ajustarlo al RETARDO_INSTRUCCION si luego lo agregás al config
    
    for (int i = 1; i < repeticiones; i++) {
        // Simulamos la ejecución reiterada
        log_info(ctx->logger, "## PID: %d - Ejecutando: NOOP -   ", contexto->pid);
        usleep(1000 * 100); // 100ms de retardo por cada repetición simulada
    }
    
    return 0;
}

t_cpu_context* iniciar_contexto(int argc, char* argv[]) {
    validar_argumentos(argc);

    t_cpu_context* ctx = malloc(sizeof(t_cpu_context));
    if (!ctx) return NULL;

    ctx->id_cpu = strdup(argv[2]);
    char* path_config = argv[1];

    printf("Se arranco el CPU correctamente con el id: %s y el path del config: %s\n", ctx->id_cpu, path_config);

    char* nombre_log = string_from_format("%s.log", ctx->id_cpu);
    ctx->logger = log_create(nombre_log, "CPU", true, LOG_LEVEL_INFO);
    free(nombre_log);

    ctx->config = config_create(path_config);
    if (ctx->config == NULL) {
        log_error(ctx->logger, "No se pudo leer el archivo de configuracion");
        log_destroy(ctx->logger);
        free(ctx->id_cpu);
        free(ctx);
        return NULL;
    }

    char* ip_scheduler = config_get_string_value(ctx->config, "IP_SCHEDULER");
    char* puerto_scheduler = config_get_string_value(ctx->config, "PUERTO_SCHEDULER");
    char* ip_memory = config_get_string_value(ctx->config, "IP_MEMORY");
    char* puerto_memory = config_get_string_value(ctx->config, "PUERTO_MEMORY");
    
    ctx->conexiones = malloc(sizeof(t_conexiones_cpu));
    ctx->conexiones->sticks = -1;

    log_info(ctx->logger, "Intentando conectar Kernel Memory...");
    ctx->conexiones->memoria = crear_conexion(ip_memory, puerto_memory);
    ctx->memory_sticks = list_create();
    if (ctx->conexiones->memoria != -1) {
        log_info(ctx->logger, "## Conectado a Kernel Memory");
        enviar_mensaje(ctx->id_cpu, ctx->conexiones->memoria);

        int op = SOLICITUD_TAM_MAX_SEGMENTO;
        send(ctx->conexiones->memoria, &op, sizeof(int), 0);
        
        // El Kernel Memory va a responder con ENVIO_TABLA_STICKS y TAM_MAX_SEGMENTO
        int recibidos = 0;
        while (recibidos < 2) {
            int resp = recibir_operacion(ctx->conexiones->memoria);
            if (resp == TAM_MAX_SEGMENTO) {
                t_list* paq = recibir_paquete(ctx->conexiones->memoria);
                int* val = list_get(paq, 0);
                ctx->tam_max_segmento = *val;
                list_destroy_and_destroy_elements(paq, free);
                log_info(ctx->logger, "Tamaño maximo de segmento recibido: %d", ctx->tam_max_segmento);
                recibidos++;
            } else if (resp == ENVIO_TABLA_STICKS) {
                int size;
                void* buf = recibir_buffer(&size, ctx->conexiones->memoria);
                int desp = 0;
                int cant;
                desp += sizeof(int); // skip size of cant
                memcpy(&cant, buf + desp, sizeof(int)); desp += sizeof(int);
                
                for (int i = 0; i < cant; i++) {
                    t_memory_stick_info* stick = malloc(sizeof(t_memory_stick_info));
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->id), buf + desp, sizeof(int)); desp += sizeof(int);
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->tamanio), buf + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->base_address), buf + desp, sizeof(uint32_t)); desp += sizeof(uint32_t);
                    
                    desp += sizeof(int); // skip size
                    memcpy(&(stick->puerto), buf + desp, sizeof(int)); desp += sizeof(int);
                    
                    int ip_len;
                    desp += sizeof(int); // skip size
                    memcpy(&ip_len, buf + desp, sizeof(int)); desp += sizeof(int);
                    
                    desp += sizeof(int); // skip size
                    stick->ip = malloc(ip_len);
                    memcpy(stick->ip, buf + desp, ip_len); desp += ip_len;
                    
                    stick->socket_fd = crear_conexion(stick->ip, string_itoa(stick->puerto)); // TODO free itoa
                    if (stick->socket_fd != -1) {
                        enviar_mensaje(ctx->id_cpu, stick->socket_fd);
                        list_add(ctx->memory_sticks, stick);
                        log_info(ctx->logger, "Conectado a Memory Stick ID %d en %s:%d", stick->id, stick->ip, stick->puerto);
                    } else {
                        log_error(ctx->logger, "Error al conectar con Memory Stick ID %d", stick->id);
                    }
                }
                free(buf);
                recibidos++;
            } else {
                log_error(ctx->logger, "Operacion no esperada en handshake de memory: %d", resp);
                break;
            }
        }
    }

    log_info(ctx->logger, "Intentando conectar Kernel Scheduler...");
    ctx->conexiones->scheduler = crear_conexion(ip_scheduler, puerto_scheduler);
    if (ctx->conexiones->scheduler != -1) {
        log_info(ctx->logger, "## Conectado a Kernel Scheduler");
        enviar_mensaje(ctx->id_cpu, ctx->conexiones->scheduler);
    }

    return ctx;
}

void limpiar_contexto(t_cpu_context* ctx) {
    if (!ctx) return;
    
    // Si la limpieza vieja existe, la reusamos.
    if (ctx->conexiones) {
        limpieza_programa(ctx->conexiones, ctx->logger, ctx->config);
    } else {
        if (ctx->logger) log_destroy(ctx->logger);
        if (ctx->config) config_destroy(ctx->config);
    }
    
    if (ctx->id_cpu) free(ctx->id_cpu);
    free(ctx);
}

int main(int argc, char* argv[]) {
    t_cpu_context* ctx = iniciar_contexto(argc, argv);
    if (!ctx) return EXIT_FAILURE;

    while (1) {
        log_info(ctx->logger, "Esperando un nuevo PID del Kernel Scheduler...");
        int process_ID = recibir_operacion(ctx->conexiones->scheduler);
        
        switch(process_ID) {
            case PID_A_EJECUTAR:
            {
                int size;
                void* buffer = recibir_buffer(&size, ctx->conexiones->scheduler);
                
                int desplazamiento = 0;
                int size_payload;
                memcpy(&size_payload, buffer + desplazamiento, sizeof(int));
                desplazamiento += sizeof(int);
                
                int pid_recibido;
                memcpy(&pid_recibido, buffer + desplazamiento, sizeof(int));
                free(buffer);

                manejar_ejecucion_proceso(pid_recibido, ctx);
                break;
            }
            case -1:
                log_error(ctx->logger, "El Kernel Scheduler se desconectó inesperadamente.");
                printf("Termino la CPU y se cerro correctamente\n");
                limpiar_contexto(ctx);
                return EXIT_FAILURE;
            default:
                log_warning(ctx->logger, "Operación desconocida recibida del Scheduler: %d", process_ID);
                limpiar_contexto(ctx);
                return EXIT_FAILURE;
        }
    }

    limpiar_contexto(ctx);
    return 0;
}
